   249→   249→function PipelineColumn({
   250→   250→  stage,
   251→   251→  leads,
   252→   252→  onLeadClick,
   253→   253→}: {
   254→   254→  stage: PipelineStage
   255→   255→  leads: Lead[]
   256→   256→  onLeadClick: (lead: Lead) => void
   257→   257→}) {
   258→   258→  const Icon = STAGE_ICONS[stage.id] || Circle
   259→   259→  const totalValue = leads.reduce((sum, l) => sum + (l.estimatedValue || 0), 0)
   260→   260→
   261→   261→  return (
   262→   262→    <div className="flex flex-col min-w-[280px] w-[280px]">
   263→   263→      {/* Column Header */}
   264→   264→      <div className="flex items-center gap-2 px-3 py-2 mb-2 rounded-lg bg-white border border-slate-200">
   265→   265→        <div className="h-7 w-7 rounded-lg flex items-center justify-center" style={{ backgroundColor: stage.color + '20' }}>
   266→   266→          <Icon className="h-4 w-4" style={{ color: stage.color }} />
   267→   267→        </div>
   268→   268→        <div className="flex-1 min-w-0">
   269→   269→          <h3 className="text-xs font-bold text-slate-900 truncate">{stage.label}</h3>
   270→   270→          <p className="text-[10px] text-muted-foreground">
   271→   271→            {leads.length} lead{leads.length !== 1 ? 's' : ''} · ₹{totalValue.toLocaleString('en-IN')}
   272→   272→          </p>
   273→   273→        </div>
   274→   274→        <Badge variant="secondary" className="h-5 text-[10px] px-1.5" style={{ backgroundColor: stage.color + '15', color: stage.color }}>
   275→   275→          {leads.length}
   276→   276→        </Badge>
   277→   277→      </div>
   278→   278→
   279→   279→      {/* Lead Cards */}
   280→   280→      <div className="flex-1 space-y-2 overflow-y-auto max-h-[calc(100vh-320px)] pr-1 custom-scrollbar">
   281→   281→        <AnimatePresence>
   282→   282→          {leads.map((lead) => (
   283→   283→            <LeadCard key={lead.id} lead={lead} onClick={() => onLeadClick(lead)} />
   284→   284→          ))}
   285→   285→        </AnimatePresence>
   286→   286→        {leads.length === 0 && (
   287→   287→          <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   288→   288→            <Icon className="h-8 w-8 mb-2 opacity-20" style={{ color: stage.color }} />
   289→   289→            <p className="text-xs">No leads here yet</p>
   290→   290→          </div>
   291→   291→        )}
   292→   292→      </div>
   293→   293→    </div>
   294→   294→  )
   295→   295→}

   299→   299→function LeadDetailPanel({
   300→   300→  lead,
   301→   301→  onClose,
   302→   302→  onRefresh,
   303→   303→}: {
   304→   304→  lead: Lead
   305→   305→  onClose: () => void
   306→   306→  onRefresh: () => void
   307→   307→}) {
   308→   308→  const { user } = useAppStore()
   309→   309→  const { toast } = useToast()
   310→   310→  const [activeTab, setActiveTab] = useState('details')
   311→   311→  const [notes, setNotes] = useState<LeadNote[]>([])
   312→   312→  const [tasks, setTasks] = useState<LeadTask[]>([])
   313→   313→  const [reminders, setReminders] = useState<Reminder[]>([])
   314→   314→  const [activities, setActivities] = useState<ActivityLogItem[]>([])
   315→   315→  const [loadingDetails, setLoadingDetails] = useState(true)
   316→   316→
   317→   317→  // Form states
   318→   318→  const [newNote, setNewNote] = useState('')
   319→   319→  const [newTask, setNewTask] = useState('')
   320→   320→  const [newTaskDue, setNewTaskDue] = useState('')
   321→   321→  const [newReminder, setNewReminder] = useState('')
   322→   322→  const [newReminderDue, setNewReminderDue] = useState('')
   323→   323→  const [selectedStatus, setSelectedStatus] = useState(lead.status)
   324→   324→  const [selectedPriority, setSelectedPriority] = useState(lead.priority)
   325→   325→  const [estimatedValue, setEstimatedValue] = useState(lead.estimatedValue?.toString() || '')
   326→   326→
   327→   327→  // Load lead details
   328→   328→  useEffect(() => {
   329→   329→    const loadDetails = async () => {
   330→   330→      setLoadingDetails(true)
   331→   331→      try {
   332→   332→        const [notesRes, tasksRes, remindersRes, activitiesRes] = await Promise.all([
   333→   333→          fetch(`/api/crm/notes?leadId=${lead.id}`),
   334→   334→          fetch(`/api/crm/tasks?leadId=${lead.id}`),
   335→   335→          fetch(`/api/crm/reminders?leadId=${lead.id}`),
   336→   336→          fetch(`/api/crm/activities?leadId=${lead.id}`),
   337→   337→        ])
   338→   338→        if (notesRes.ok) setNotes((await notesRes.json()).notes || [])
   339→   339→        if (tasksRes.ok) setTasks((await tasksRes.json()).tasks || [])
   340→   340→        if (remindersRes.ok) setReminders((await remindersRes.json()).reminders || [])
   341→   341→        if (activitiesRes.ok) setActivities((await activitiesRes.json()).activities || [])
   342→   342→      } catch {
   343→   343→        // Silent fail
   344→   344→      } finally {
   345→   345→        setLoadingDetails(false)
   346→   346→      }
   347→   347→    }
   348→   348→    loadDetails()
   349→   349→  }, [lead.id])
   350→   350→
   351→   351→  // Update lead
   352→   352→  const handleUpdateLead = async () => {
   353→   353→    try {
   354→   354→      const res = await fetch('/api/crm/pipeline', {
   355→   355→        method: 'PATCH',
   356→   356→        headers: { 'Content-Type': 'application/json' },
   357→   357→        body: JSON.stringify({
   358→   358→          leadId: lead.id,
   359→   359→          status: selectedStatus,
   360→   360→          priority: selectedPriority,
   361→   361→          estimatedValue: estimatedValue ? parseFloat(estimatedValue) : null,
   362→   362→          userId: user?.id || 'demo',
   363→   363→        }),
   364→   364→      })
   365→   365→      if (res.ok) {
   366→   366→        toast({ title: 'Lead updated!' })
   367→   367→        onRefresh()
   368→   368→      }
   369→   369→    } catch {
   370→   370→      toast({ title: 'Failed to update lead', variant: 'destructive' })
   371→   371→    }
   372→   372→  }
   373→   373→
   374→   374→  // Add note
   375→   375→  const handleAddNote = async () => {
   376→   376→    if (!newNote.trim()) return
   377→   377→    try {
   378→   378→      const res = await fetch('/api/crm/notes', {
   379→   379→        method: 'POST',
   380→   380→        headers: { 'Content-Type': 'application/json' },
   381→   381→        body: JSON.stringify({ leadId: lead.id, userId: user?.id || 'demo', content: newNote.trim() }),
   382→   382→      })
   383→   383→      if (res.ok) {
   384→   384→        const data = await res.json()
   385→   385→        setNotes(prev => [data.note, ...prev])
   386→   386→        setNewNote('')
   387→   387→        toast({ title: 'Note added!' })
   388→   388→      }
   389→   389→    } catch {
   390→   390→      toast({ title: 'Failed to add note', variant: 'destructive' })
   391→   391→    }
   392→   392→  }
   393→   393→
   394→   394→  // Add task
   395→   395→  const handleAddTask = async () => {
   396→   396→    if (!newTask.trim()) return
   397→   397→    try {
   398→   398→      const res = await fetch('/api/crm/tasks', {
   399→   399→        method: 'POST',
   400→   400→        headers: { 'Content-Type': 'application/json' },
   401→   401→        body: JSON.stringify({
   402→   402→          leadId: lead.id,
   403→   403→          userId: user?.id || 'demo',
   404→   404→          title: newTask.trim(),
   405→   405→          dueDate: newTaskDue || null,
   406→   406→        }),
   407→   407→      })
   408→   408→      if (res.ok) {
   409→   409→        const data = await res.json()
   410→   410→        setTasks(prev => [data.task, ...prev])
   411→   411→        setNewTask('')
   412→   412→        setNewTaskDue('')
   413→   413→        toast({ title: 'Task added!' })
   414→   414→      }
   415→   415→    } catch {
   416→   416→      toast({ title: 'Failed to add task', variant: 'destructive' })
   417→   417→    }
   418→   418→  }
   419→   419→
   420→   420→  // Toggle task
   421→   421→  const handleToggleTask = async (taskId: string, completed: boolean) => {
   422→   422→    try {
   423→   423→      const res = await fetch('/api/crm/tasks', {
   424→   424→        method: 'PATCH',
   425→   425→        headers: { 'Content-Type': 'application/json' },
   426→   426→        body: JSON.stringify({ taskId, completed: !completed }),
   427→   427→      })
   428→   428→      if (res.ok) {
   429→   429→        setTasks(prev => prev.map(t => t.id === taskId ? { ...t, completed: !completed, completedAt: !completed ? new Date().toISOString() : null } : t))
   430→   430→        toast({ title: completed ? 'Task reopened' : 'Task completed!' })
   431→   431→      }
   432→   432→    } catch {
   433→   433→      toast({ title: 'Failed to update task', variant: 'destructive' })
   434→   434→    }
   435→   435→  }
   436→   436→
   437→   437→  // Delete task
   438→   438→  const handleDeleteTask = async (taskId: string) => {
   439→   439→    try {
   440→   440→      const res = await fetch(`/api/crm/tasks?taskId=${taskId}`, { method: 'DELETE' })
   441→   441→      if (res.ok) {
   442→   442→        setTasks(prev => prev.filter(t => t.id !== taskId))
   443→   443→        toast({ title: 'Task deleted' })
   444→   444→      }
   445→   445→    } catch {
   446→   446→      toast({ title: 'Failed to delete task', variant: 'destructive' })
   447→   447→    }
   448→   448→  }
   449→   449→
   450→   450→  // Add reminder
   451→   451→  const handleAddReminder = async () => {
   452→   452→    if (!newReminder.trim() || !newReminderDue) return
   453→   453→    try {
   454→   454→      const res = await fetch('/api/crm/reminders', {
   455→   455→        method: 'POST',
   456→   456→        headers: { 'Content-Type': 'application/json' },
   457→   457→        body: JSON.stringify({
   458→   458→          leadId: lead.id,
   459→   459→          userId: user?.id || 'demo',
   460→   460→          title: newReminder.trim(),
   461→   461→          dueDate: newReminderDue,
   462→   462→        }),
   463→   463→      })
   464→   464→      if (res.ok) {
   465→   465→        const data = await res.json()
   466→   466→        setReminders(prev => [data.reminder, ...prev])
   467→   467→        setNewReminder('')
   468→   468→        setNewReminderDue('')
   469→   469→        toast({ title: 'Reminder set!' })
   470→   470→      }
   471→   471→    } catch {
   472→   472→      toast({ title: 'Failed to add reminder', variant: 'destructive' })
   473→   473→    }
   474→   474→  }
   475→   475→
   476→   476→  // Complete reminder
   477→   477→  const handleCompleteReminder = async (reminderId: string) => {
   478→   478→    try {
   479→   479→      const res = await fetch('/api/crm/reminders', {
   480→   480→        method: 'PATCH',
   481→   481→        headers: { 'Content-Type': 'application/json' },
   482→   482→        body: JSON.stringify({ reminderId, completed: true }),
