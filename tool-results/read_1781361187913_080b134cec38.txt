    92→function getWebsiteStatusConfig(status: string | null | undefined) {
    93→  switch (status) {
    94→    case 'NO_WEBSITE':
    95→      return { label: 'No Website', color: 'text-red-600', bg: 'bg-red-50', border: 'border-red-200', icon: Unplug }
    96→    case 'SOCIAL_ONLY':
    97→      return { label: 'Social Only', color: 'text-amber-600', bg: 'bg-amber-50', border: 'border-amber-200', icon: Globe }
    98→    case 'HAS_WEBSITE':
    99→      return { label: 'Has Website', color: 'text-emerald-600', bg: 'bg-emerald-50', border: 'border-emerald-200', icon: Globe }
   100→    default:
   101→      return { label: 'Unknown', color: 'text-slate-600', bg: 'bg-slate-50', border: 'border-slate-200', icon: Globe }
   102→  }
   103→}
   104→
   105→function getScoreColor(score: number | null | undefined): string {
   106→  if (!score) return 'text-slate-400'
   107→  if (score >= 80) return 'text-emerald-600'
   108→  if (score >= 60) return 'text-amber-600'
   109→  if (score >= 40) return 'text-orange-600'
   110→  return 'text-red-600'
   111→}
   112→
   113→function getScoreBg(score: number | null | undefined): string {
   114→  if (!score) return 'bg-slate-100'
   115→  if (score >= 80) return 'bg-emerald-50'
   116→  if (score >= 60) return 'bg-amber-50'
   117→  if (score >= 40) return 'bg-orange-50'
   118→  return 'bg-red-50'
   119→}

   507→                  <div className="flex items-center justify-between p-3 rounded-xl bg-slate-50 dark:bg-slate-800">
   508→                    <div className="flex items-center gap-3">
   509→                      <div className="h-9 w-9 rounded-lg bg-slate-100 dark:bg-slate-700 flex items-center justify-center">
   510→                        <Globe className="h-4 w-4 text-slate-600 dark:text-slate-400" />
   511→                      </div>
   512→                      <div>
   513→                        <p className="text-xs text-muted-foreground">Website</p>
   514→                        <p className="text-sm font-medium">{business.website}</p>
   515→                      </div>
   516→                    </div>
   517→                    <Button size="sm" variant="outline" className="text-xs" onClick={() => window.open(business.website!, '_blank')}>
   518→                      <ExternalLink className="h-3 w-3 mr-1" /> Visit
   519→                    </Button>
   520→                  </div>
   521→                )}
   522→                {business.address && (
   523→                  <div className="p-3 rounded-xl bg-slate-50 dark:bg-slate-800">
   524→                    <div className="flex items-center gap-3">
   525→                      <div className="h-9 w-9 rounded-lg bg-red-50 dark:bg-red-500/10 flex items-center justify-center">
   526→                        <MapPin className="h-4 w-4 text-red-600 dark:text-red-400" />
   527→                      </div>
   528→                      <div>
   529→                        <p className="text-xs text-muted-foreground">Address</p>
   530→                        <p className="text-sm font-medium">{business.address}</p>
   531→                        <p className="text-xs text-muted-foreground">
   532→                          {[business.city, business.state, business.country].filter(Boolean).join(', ')}
   533→                        </p>
   534→                      </div>
   535→                    </div>
   536→                  </div>
   537→                )}
   538→
   539→                {!business.phone && !business.email && !business.website && (
   540→                  <div className="flex flex-col items-center py-8 text-center">
   541→                    <AlertCircle className="h-8 w-8 text-slate-300 mb-2" />
   542→                    <p className="text-sm text-muted-foreground">No contact info available</p>
   543→                    <p className="text-xs text-muted-foreground">Try searching for more details</p>
   544→                  </div>
   545→                )}
   546→              </div>
   547→            </TabsContent>
   548→
   549→            {/* Scoring Tab */}
   550→            <TabsContent value="scoring" className="p-6 space-y-5 mt-0">
   551→              {/* Lead Score Breakdown */}
   552→              <div className="space-y-3">
   553→                <h3 className="text-sm font-semibold text-slate-900 dark:text-white">Lead Score Breakdown</h3>
   554→
   555→                <div className={`rounded-xl p-5 text-center ${getScoreBg(business.leadScore)}`}>
   556→                  <p className={`text-4xl font-bold ${getScoreColor(business.leadScore)}`}>
   557→                    {business.leadScore ?? '—'}
   558→                  </p>
   559→                  <p className="text-sm text-muted-foreground mt-1">out of 100</p>
   560→                  <div className="w-full bg-white/50 dark:bg-slate-700/50 rounded-full h-2 mt-3">
   561→                    <div
   562→                      className={`h-2 rounded-full transition-all ${
   563→                        (business.leadScore ?? 0) >= 80 ? 'bg-emerald-500' :
   564→                        (business.leadScore ?? 0) >= 60 ? 'bg-amber-500' :
   565→                        (business.leadScore ?? 0) >= 40 ? 'bg-orange-500' : 'bg-red-500'
   566→                      }`}
   567→                      style={{ width: `${business.leadScore ?? 0}%` }}
   568→                    />
   569→                  </div>
   570→                </div>
   571→
   572→                {/* Score Factors */}
   573→                <div className="space-y-2">
   574→                  {[
   575→                    { label: 'No Website', value: !business.hasWebsite, points: 30 },
   576→                    { label: 'Social Only', value: business.websiteStatus === 'SOCIAL_ONLY', points: 20 },
   577→                    { label: 'Has Phone', value: !!business.phone, points: 10 },
   578→                    { label: 'Has Email', value: !!business.email, points: 10 },
   579→                    { label: 'Google Rating < 4', value: (business.googleRating ?? 5) < 4, points: 15 },
   580→                    { label: 'Social Presence', value: business.socialPresence > 0, points: business.socialPresence * 5 },
   581→                  ].map((factor) => (
   582→                    <div key={factor.label} className="flex items-center justify-between p-2.5 rounded-lg bg-slate-50 dark:bg-slate-800">
   583→                      <div className="flex items-center gap-2">
   584→                        {factor.value ? (
   585→                          <CheckCircle2 className="h-4 w-4 text-emerald-500" />
   586→                        ) : (
   587→                          <X className="h-4 w-4 text-slate-300" />
   588→                        )}
   589→                        <span className="text-sm">{factor.label}</span>
   590→                      </div>
   591→                      <Badge variant="secondary" className={`text-xs ${factor.value ? 'bg-emerald-50 text-emerald-700' : 'bg-slate-100 text-slate-400'}`}>
   592→                        +{factor.points}
   593→                      </Badge>
   594→                    </div>
   595→                  ))}
   596→                </div>
   597→              </div>
   598→
   599→              <Separator />
   600→
   601→              {/* Opportunity Score */}
   602→              <div className="space-y-3">
   603→                <h3 className="text-sm font-semibold text-slate-900 dark:text-white">Opportunity Score</h3>
   604→                <div className={`rounded-xl p-5 text-center ${getScoreBg(business.opportunityScore)}`}>
   605→                  <p className={`text-4xl font-bold ${getScoreColor(business.opportunityScore)}`}>
   606→                    {business.opportunityScore ?? '—'}
   607→                  </p>
   608→                  <p className="text-sm text-muted-foreground mt-1">out of 100</p>
   609→                  <div className="w-full bg-white/50 dark:bg-slate-700/50 rounded-full h-2 mt-3">
   610→                    <div
   611→                      className={`h-2 rounded-full transition-all ${
   612→                        (business.opportunityScore ?? 0) >= 80 ? 'bg-emerald-500' :
   613→                        (business.opportunityScore ?? 0) >= 60 ? 'bg-amber-500' :
   614→                        (business.opportunityScore ?? 0) >= 40 ? 'bg-orange-500' : 'bg-red-500'
   615→                      }`}
   616→                      style={{ width: `${business.opportunityScore ?? 0}%` }}
   617→                    />
   618→                  </div>
   619→                </div>
   620→              </div>
   621→
   622→              {/* Estimated Revenue */}
   623→              {business.estimatedMonthlyRevenue && (
   624→                <>
   625→                  <Separator />
   626→                  <div className="space-y-2">
   627→                    <h3 className="text-sm font-semibold text-slate-900 dark:text-white">Estimated Revenue</h3>
   628→                    <div className="rounded-xl p-4 bg-amber-50 dark:bg-amber-500/10 text-center">
   629→                      <p className="text-2xl font-bold text-amber-700 dark:text-amber-400">
   630→                        ₹{business.estimatedMonthlyRevenue.toLocaleString()}
   631→                      </p>
   632→                      <p className="text-xs text-muted-foreground">per month (estimated)</p>
   633→                    </div>
   634→                  </div>
   635→                </>
   636→              )}
   637→            </TabsContent>
   638→
   639→            {/* Social Tab */}
   640→            <TabsContent value="social" className="p-6 space-y-5 mt-0">
   641→              <div className="space-y-3">
   642→                <h3 className="text-sm font-semibold text-slate-900 dark:text-white">Social Media Presence</h3>
   643→
   644→                {/* Social Score */}
   645→                <div className="rounded-xl p-4 bg-slate-50 dark:bg-slate-800 flex items-center justify-between">
   646→                  <div>
   647→                    <p className="text-sm font-medium">Social Presence Score</p>
   648→                    <p className="text-xs text-muted-foreground">Based on connected platforms</p>
   649→                  </div>
   650→                  <div className="text-2xl font-bold text-amber-600">{business.socialPresence}/3</div>
   651→                </div>
   652→
   653→                {/* Platform Cards */}
   654→                {[
   655→                  { name: 'Facebook', url: business.facebookUrl, icon: Facebook, color: 'text-blue-600', bg: 'bg-blue-50 dark:bg-blue-500/10' },
   656→                  { name: 'Instagram', url: business.instagramUrl, icon: Instagram, color: 'text-pink-600', bg: 'bg-pink-50 dark:bg-pink-500/10' },
   657→                  { name: 'LinkedIn', url: business.linkedinUrl, icon: Linkedin, color: 'text-blue-700', bg: 'bg-blue-50 dark:bg-blue-500/10' },
   658→                ].map((platform) => (
   659→                  <div key={platform.name} className={`rounded-xl p-4 border ${platform.url ? 'border-slate-200 dark:border-slate-700' : 'border-dashed border-slate-200 dark:border-slate-700'}`}>
   660→                    <div className="flex items-center justify-between">
   661→                      <div className="flex items-center gap-3">
   662→                        <div className={`h-10 w-10 rounded-xl ${platform.url ? platform.bg : 'bg-slate-100 dark:bg-slate-800'} flex items-center justify-center`}>
   663→                          <platform.icon className={`h-5 w-5 ${platform.url ? platform.color : 'text-slate-300'}`} />
   664→                        </div>
   665→                        <div>
   666→                          <p className="text-sm font-medium">{platform.name}</p>
   667→                          {platform.url ? (
   668→                            <p className="text-xs text-amber-600 truncate max-w-[200px]">{platform.url}</p>
   669→                          ) : (
   670→                            <p className="text-xs text-muted-foreground">Not found</p>
   671→                          )}
   672→                        </div>
   673→                      </div>
   674→                      {platform.url && (
   675→                        <Button size="sm" variant="ghost" className="text-xs" onClick={() => window.open(platform.url!, '_blank')}>
   676→                          <ExternalLink className="h-3.5 w-3.5" />
   677→                        </Button>
   678→                      )}
   679→                    </div>
   680→                  </div>
   681→                ))}
   682→              </div>
   683→            </TabsContent>
   684→          </ScrollArea>
   685→        </Tabs>
   686→      </SheetContent>
   687→    </Sheet>
   688→  )
   689→}
