   132→function formatDate(dateStr: string): string {
   133→  try {
   134→    return new Date(dateStr).toLocaleDateString('en-US', {
   135→      year: 'numeric',
   136→      month: 'short',
   137→      day: 'numeric',
   138→    })
   139→  } catch {
   140→    return dateStr
   141→  }
   142→}
   143→
   144→// formatCurrency replaced by useCurrency hook inside the component
   145→
   146→function formatExpiry(month: number, year: number): string {
   147→  return `${String(month).padStart(2, '0')}/${year}`
   148→}
