    61→function formatRelativeTime(dateStr: string): string {
    62→  const date = new Date(dateStr)
    63→  const now = new Date()
    64→  const diffMs = now.getTime() - date.getTime()
    65→  const diffSeconds = Math.floor(diffMs / 1000)
    66→  const diffMinutes = Math.floor(diffSeconds / 60)
    67→  const diffHours = Math.floor(diffMinutes / 60)
    68→  const diffDays = Math.floor(diffHours / 24)
    69→
    70→  if (diffSeconds < 60) return 'Just now'
    71→  if (diffMinutes < 60) return `${diffMinutes}m ago`
    72→  if (diffHours < 24) return `${diffHours}h ago`
    73→  if (diffDays < 7) return `${diffDays}d ago`
    74→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
    75→}

   237→export function NotificationsView() {
   238→  const { user } = useAppStore()
   239→  const { toast } = useToast()
   240→
   241→  // Notification state
   242→  const [notifications, setNotifications] = useState<ApiNotification[]>([])
   243→  const [apiUnreadCount, setApiUnreadCount] = useState<number>(0)
   244→  const [loading, setLoading] = useState(true)
   245→  const [preferencesLoading, setPreferencesLoading] = useState(true)
   246→  const [error, setError] = useState<string | null>(null)
   247→  const [activeTab, setActiveTab] = useState<string>('all')
   248→  const [markingRead, setMarkingRead] = useState<string | null>(null)
   249→
   250→  // Preferences state (loaded from API)
   251→  const [preferences, setPreferences] = useState<NotificationPreference[]>(DEFAULT_PREFERENCES)
   252→
   253→  // Fetch notifications
   254→  const fetchNotifications = useCallback(async () => {
   255→    if (!user?.id) return
   256→    setLoading(true)
   257→    setError(null)
   258→    try {
   259→      const res = await fetch('/api/user/notifications?limit=50', {
   260→        credentials: 'include',
   261→      })
   262→      if (!res.ok) {
   263→        const body = await res.json().catch(() => ({}))
   264→        throw new Error(body.error || 'Failed to fetch notifications')
   265→      }
   266→      const json = await res.json()
   267→      setNotifications(json.notifications ?? [])
   268→      setApiUnreadCount(json.unreadCount ?? 0)
   269→    } catch (err: any) {
   270→      console.error('Failed to fetch notifications:', err)
   271→      setError(err.message || 'Failed to load notifications')
   272→      toast({
   273→        title: 'Error',
   274→        description: 'Failed to load notifications. Please try again.',
   275→        variant: 'destructive',
   276→      })
   277→    } finally {
   278→      setLoading(false)
   279→    }
   280→  }, [user?.id, toast])
   281→
   282→  // Fetch preferences from API
   283→  const fetchPreferences = useCallback(async () => {
   284→    if (!user?.id) return
   285→    setPreferencesLoading(true)
   286→    try {
   287→      const res = await fetch('/api/user/notifications/preferences', {
   288→        credentials: 'include',
   289→      })
   290→      if (!res.ok) {
   291→        throw new Error('Failed to fetch preferences')
   292→      }
   293→      const json = await res.json()
   294→      // Map API response to our local state structure
   295→      const apiPrefs = json.preferences as Record<string, { id: string; itemKey: string; enabled: boolean }[]>
   296→
   297→      setPreferences((prev) =>
   298→        prev.map((cat) => {
   299→          const apiItems = apiPrefs[cat.category]
   300→          if (!apiItems) return cat
   301→          return {
   302→            ...cat,
   303→            items: cat.items.map((item) => {
   304→              const apiItem = apiItems.find((a) => a.itemKey === item.itemKey)
   305→              if (apiItem) {
   306→                return { ...item, id: apiItem.id, enabled: apiItem.enabled }
   307→              }
   308→              return item
   309→            }),
   310→          }
   311→        })
   312→      )
   313→    } catch (err) {
   314→      console.error('Failed to fetch preferences:', err)
   315→      // Keep defaults on error
   316→    } finally {
   317→      setPreferencesLoading(false)
   318→    }
   319→  }, [user?.id])
   320→
   321→  useEffect(() => {
   322→    fetchNotifications()
   323→    fetchPreferences()
   324→  }, [fetchNotifications, fetchPreferences])
   325→
   326→  // Derived state
   327→  const unreadCount = useMemo(() => notifications.filter((n) => !n.read).length, [notifications])
   328→
   329→  const filteredNotifications = useMemo(() => {
   330→    switch (activeTab) {
   331→      case 'unread':
   332→        return notifications.filter((n) => !n.read)
   333→      default:
   334→        return notifications
   335→    }
   336→  }, [notifications, activeTab])
   337→
   338→  // Actions
   339→  const markAsRead = async (id: string) => {
   340→    if (!user?.id) return
   341→    // Optimistic update
   342→    const prev = notifications
   343→    setNotifications((ns) =>
   344→      ns.map((n) => (n.id === id ? { ...n, read: true } : n))
   345→    )
   346→    setMarkingRead(id)
   347→    try {
   348→      const res = await fetch('/api/user/notifications', {
   349→        method: 'PUT',
   350→        headers: { 'Content-Type': 'application/json' },
   351→        credentials: 'include',
   352→        body: JSON.stringify({ notificationId: id }),
   353→      })
   354→      if (!res.ok) {
   355→        throw new Error('Failed to mark as read')
   356→      }
   357→    } catch (err) {
   358→      console.error('Failed to mark as read:', err)
   359→      // Rollback
   360→      setNotifications(prev)
   361→      toast({
   362→        title: 'Error',
   363→        description: 'Failed to mark notification as read.',
   364→        variant: 'destructive',
   365→      })
   366→    } finally {
   367→      setMarkingRead(null)
   368→    }
   369→  }
   370→
   371→  const markAllAsRead = async () => {
   372→    if (!user?.id) return
   373→    // Optimistic update
   374→    const prev = notifications
   375→    setNotifications((ns) => ns.map((n) => ({ ...n, read: true })))
   376→    try {
   377→      const res = await fetch('/api/user/notifications', {
   378→        method: 'PUT',
   379→        headers: { 'Content-Type': 'application/json' },
   380→        credentials: 'include',
   381→        body: JSON.stringify({ markAllRead: true }),
   382→      })
   383→      if (!res.ok) {
   384→        throw new Error('Failed to mark all as read')
   385→      }
   386→      toast({
   387→        title: 'Success',
   388→        description: 'All notifications marked as read.',
   389→      })
   390→    } catch (err) {
   391→      console.error('Failed to mark all as read:', err)
   392→      // Rollback
   393→      setNotifications(prev)
   394→      toast({
   395→        title: 'Error',
   396→        description: 'Failed to mark all notifications as read.',
   397→        variant: 'destructive',
   398→      })
   399→    }
   400→  }
   401→
   402→  const dismissNotification = async (id: string) => {
   403→    // Optimistic update
   404→    const prev = notifications
   405→    setNotifications((ns) => ns.filter((n) => n.id !== id))
   406→    try {
   407→      const res = await fetch(`/api/user/notifications?notificationId=${id}`, {
   408→        method: 'DELETE',
   409→        credentials: 'include',
   410→      })
   411→      if (!res.ok) {
   412→        throw new Error('Failed to delete notification')
   413→      }
   414→      toast({
   415→        title: 'Dismissed',
   416→        description: 'Notification removed.',
   417→      })
   418→    } catch (err) {
   419→      console.error('Failed to delete notification:', err)
   420→      // Rollback
   421→      setNotifications(prev)
   422→      toast({
   423→        title: 'Error',
   424→        description: 'Failed to dismiss notification.',
   425→        variant: 'destructive',
   426→      })
   427→    }
   428→  }
   429→
   430→  const togglePreference = async (categoryId: string, itemKey: string) => {
   431→    // Find current value
   432→    const category = preferences.find((c) => c.category === categoryId)
   433→    const item = category?.items.find((i) => i.itemKey === itemKey)
   434→    if (!item) return
   435→
   436→    const newEnabled = !item.enabled
   437→
   438→    // Optimistic update
   439→    setPreferences((prev) =>
   440→      prev.map((cat) =>
   441→        cat.category === categoryId
   442→          ? {
   443→              ...cat,
   444→              items: cat.items.map((i) =>
   445→                i.itemKey === itemKey ? { ...i, enabled: newEnabled, saving: true } : i
   446→              ),
   447→            }
   448→          : cat
   449→      )
   450→    )
   451→
   452→    try {
   453→      const res = await fetch('/api/user/notifications/preferences', {
   454→        method: 'PUT',
   455→        headers: { 'Content-Type': 'application/json' },
   456→        credentials: 'include',
   457→        body: JSON.stringify({ category: categoryId, itemKey, enabled: newEnabled }),
   458→      })
   459→      if (!res.ok) {
   460→        throw new Error('Failed to update preference')
   461→      }
   462→      toast({
   463→        title: 'Preference Updated',
   464→        description: `${item.label} ${newEnabled ? 'enabled' : 'disabled'}.`,
   465→      })
   466→    } catch (err) {
   467→      console.error('Failed to update preference:', err)
   468→      // Rollback
   469→      setPreferences((prev) =>
   470→        prev.map((cat) =>
   471→          cat.category === categoryId
   472→            ? {
   473→                ...cat,
   474→                items: cat.items.map((i) =>
   475→                  i.itemKey === itemKey ? { ...i, enabled: !newEnabled, saving: false } : i
   476→                ),
   477→              }
   478→            : cat
   479→        )
   480→      )
   481→      toast({
   482→        title: 'Error',
   483→        description: 'Failed to update notification preference.',
   484→        variant: 'destructive',
   485→      })
   486→      return
   487→    }
   488→
   489→    // Clear saving state
   490→    setPreferences((prev) =>
   491→      prev.map((cat) =>
   492→        cat.category === categoryId
   493→          ? {
   494→              ...cat,
   495→              items: cat.items.map((i) =>
   496→                i.itemKey === itemKey ? { ...i, saving: false } : i
   497→              ),
   498→            }
   499→          : cat
   500→      )
   501→    )
   502→  }
   503→
   504→  // ─── Loading State ──────────────────────────────────────────────────────
   505→
   506→  if (loading) return <NotificationsSkeleton />
   507→
   508→  // ─── Error State ────────────────────────────────────────────────────────
   509→
   510→  if (error && notifications.length === 0) {
   511→    return (
   512→      <div className="space-y-6 max-w-3xl">
   513→        <div className="flex items-center gap-3">
   514→          <div className="h-10 w-10 rounded-xl bg-amber-50 flex items-center justify-center">
   515→            <Bell className="h-5 w-5 text-amber-600" />
   516→          </div>
   517→          <div>
   518→            <h1 className="text-2xl font-bold text-slate-900">Notifications</h1>
   519→            <p className="text-sm text-muted-foreground">Manage your notifications</p>
   520→          </div>
   521→        </div>
   522→        <Card className="border-0 shadow-sm">
   523→          <CardContent className="p-12 flex flex-col items-center text-center">
   524→            <div className="h-14 w-14 rounded-2xl bg-red-50 flex items-center justify-center mb-4">
   525→              <AlertCircle className="h-7 w-7 text-red-500" />
   526→            </div>
   527→            <h3 className="text-lg font-semibold text-slate-700 mb-1">Failed to load notifications</h3>
   528→            <p className="text-sm text-muted-foreground mb-4">{error}</p>
   529→            <Button onClick={fetchNotifications} variant="outline">
   530→              <RefreshCw className="h-4 w-4 mr-2" />
   531→              Try Again
   532→            </Button>
   533→          </CardContent>
   534→        </Card>
   535→      </div>
   536→    )
   537→  }
   538→
   539→  // ─── Render ───────────────────────────────────────────────────────────────
   540→
   541→  return (
   542→    <div className="space-y-6 max-w-3xl">
   543→      {/* Header */}
   544→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   545→        <div className="flex items-center gap-3">
   546→          <div className="h-10 w-10 rounded-xl bg-amber-50 flex items-center justify-center">
   547→            <Bell className="h-5 w-5 text-amber-600" />
   548→          </div>
   549→          <div>
   550→            <h1 className="text-2xl font-bold text-slate-900">Notifications</h1>
   551→            <p className="text-sm text-muted-foreground">
   552→              {unreadCount > 0
   553→                ? `You have ${unreadCount} unread notification${unreadCount !== 1 ? 's' : ''}`
   554→                : 'All caught up!'}
   555→            </p>
   556→          </div>
   557→        </div>
   558→        <div className="flex items-center gap-2">
   559→          {unreadCount > 0 && (
   560→            <Button
   561→              variant="outline"
   562→              size="sm"
   563→              onClick={markAllAsRead}
   564→              className="text-amber-600 border-amber-200 hover:bg-amber-50"
   565→            >
   566→              <CheckCheck className="h-4 w-4 mr-1.5" />
   567→              Mark All Read
   568→            </Button>
   569→          )}
   570→          <Badge variant="secondary" className="bg-slate-100 text-slate-600 text-xs px-2.5">
   571→            {notifications.length} total
   572→          </Badge>
   573→        </div>
   574→      </div>
   575→
   576→      {/* Tabs: All / Unread */}
   577→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   578→        <TabsList className="bg-slate-100 h-10 p-1">
   579→          <TabsTrigger value="all" className="text-sm data-[state=active]:bg-white data-[state=active]:shadow-sm">
   580→            All
   581→          </TabsTrigger>
   582→          <TabsTrigger value="unread" className="text-sm data-[state=active]:bg-white data-[state=active]:shadow-sm relative">
   583→            Unread
   584→            {unreadCount > 0 && (
   585→              <span className="ml-1.5 inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-amber-500 text-[10px] font-bold text-white px-1">
   586→                {unreadCount}
   587→              </span>
   588→            )}
   589→          </TabsTrigger>
