   151→   151→function formatCurrency(value: number): string {
   152→   152→  if (value >= 1000000) return `$${(value / 1000000).toFixed(1)}M`
   153→   153→  if (value >= 1000) return `$${(value / 1000).toFixed(0)}k`
   154→   154→  return `$${value.toLocaleString()}`
   155→   155→}
   156→   156→
   157→   157→function getScoreColor(score: number | null | undefined): { bg: string; text: string } {
   158→   158→  if (score === null || score === undefined) return { bg: 'bg-slate-100', text: 'text-slate-400' }
   159→   159→  if (score >= 70) return { bg: 'bg-emerald-50', text: 'text-emerald-600' }
   160→   160→  if (score >= 40) return { bg: 'bg-amber-50', text: 'text-amber-600' }
   161→   161→  return { bg: 'bg-red-50', text: 'text-red-500' }
   162→   162→}
