   129→// ─── Helpers ──────────────────────────────────────────────────────────────────

   131→function formatDate(dateStr: string): string {
   132→  try {
   133→    return new Date(dateStr).toLocaleDateString('en-US', {
   134→      year: 'numeric',
   135→      month: 'short',
   136→      day: 'numeric',
   137→    })
   138→  } catch {
   139→    return dateStr
   140→  }
   141→}
   142→
   143→function formatCurrency(amount: number): string {
   144→  return new Intl.NumberFormat('en-US', {
   145→    style: 'currency',
   146→    currency: 'USD',
   147→  }).format(amount)
   148→}
   149→
   150→function formatExpiry(month: number, year: number): string {
   151→  return `${String(month).padStart(2, '0')}/${year}`
   152→}
