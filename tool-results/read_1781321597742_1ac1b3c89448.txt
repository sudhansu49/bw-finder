   158→function PackageCard({ pkg, onSelect }: { pkg: ProposalPackage; onSelect: () => void }) {
   159→  const config = tierConfig[pkg.tier]
   160→  const Icon = config.icon
   161→  const includedFeatures = pkg.features.filter(f => f.included)
   162→  const excludedFeatures = pkg.features.filter(f => !f.included)
   163→
   164→  return (
   165→    <motion.div
   166→      initial={{ opacity: 0, y: 20 }}
   167→      animate={{ opacity: 1, y: 0 }}
   168→      transition={{ delay: pkg.tier === 'basic' ? 0 : pkg.tier === 'professional' ? 0.1 : 0.2 }}
   169→      className={`relative rounded-2xl border-2 ${config.border} bg-white overflow-hidden flex flex-col ${pkg.recommended ? 'shadow-lg shadow-amber-100' : 'shadow-sm'}`}
   170→    >
   171→      {pkg.recommended && (
   172→        <div className="bg-gradient-to-r from-amber-500 to-orange-500 text-white text-center py-1.5 text-xs font-bold tracking-wider uppercase">
   173→          ★ Recommended
   174→        </div>
   175→      )}
   176→
   177→      {/* Header */}
   178→      <div className={`p-5 ${config.headerBg} bg-gradient-to-br ${config.gradient}`}>
   179→        <div className="flex items-center gap-2 mb-2">
   180→          <div className={`h-8 w-8 rounded-lg ${config.accentBg} flex items-center justify-center`}>
   181→            <Icon className={`h-4 w-4 ${config.iconColor}`} />
   182→          </div>
   183→          <h3 className="text-lg font-bold uppercase tracking-wide text-slate-900">{pkg.name}</h3>
   184→        </div>
   185→        <p className="text-xs text-muted-foreground mb-4 line-clamp-2">{pkg.description}</p>
   186→        
   187→        {/* Price */}
   188→        <div className="flex items-baseline gap-2">
   189→          {pkg.originalPrice && (
   190→            <span className="text-sm text-slate-400 line-through">${pkg.originalPrice.toLocaleString()}</span>
   191→          )}
   192→          <span className={`text-3xl font-extrabold ${config.priceColor}`}>${pkg.price.toLocaleString()}</span>
   193→        </div>
   194→        <p className="text-xs text-muted-foreground mt-1">one-time setup fee</p>
   195→      </div>
   196→
   197→      {/* Timeline */}
   198→      <div className={`mx-5 mt-4 mb-3 px-3 py-2 rounded-lg ${config.accentBg} flex items-center justify-center gap-2`}>
   199→        <Clock className={`h-3.5 w-3.5 ${config.iconColor}`} />
   200→        <span className={`text-xs font-semibold ${config.accent}`}>Delivery: {pkg.timeline}</span>
   201→      </div>
   202→
   203→      {/* Features */}
   204→      <div className="px-5 flex-1">
   205→        <ul className="space-y-1.5">
   206→          {includedFeatures.map((f, i) => (
   207→            <li key={i} className="flex items-start gap-2">
   208→              <CheckCircle2 className={`h-3.5 w-3.5 mt-0.5 shrink-0 ${config.checkColor}`} />
   209→              <span className={`text-xs ${f.highlight ? 'font-semibold text-slate-800' : 'text-slate-600'}`}>{f.name}</span>
   210→            </li>
   211→          ))}
   212→          {excludedFeatures.slice(0, 3).map((f, i) => (
   213→            <li key={`ex-${i}`} className="flex items-start gap-2">
   214→              <X className="h-3.5 w-3.5 mt-0.5 shrink-0 text-slate-300" />
   215→              <span className="text-xs text-slate-400 line-through">{f.name}</span>
   216→            </li>
   217→          ))}
   218→        </ul>
   219→      </div>
   220→
   221→      {/* CTA */}
   222→      <div className="p-5 pt-4">
   223→        <Button
   224→          className={`w-full ${pkg.recommended ? 'bg-amber-500 hover:bg-amber-600 text-white' : 'bg-slate-800 hover:bg-slate-900 text-white'}`}
   225→          onClick={onSelect}
   226→        >
   227→          Select {pkg.name}
   228→          <ArrowRight className="ml-2 h-4 w-4" />
   229→        </Button>
   230→      </div>
   231→    </motion.div>
   232→  )
   233→}

   236→
   237→export function ProposalView() {
   238→  const { user } = useAppStore()
   239→  const { toast } = useToast()
   240→  const [businesses, setBusinesses] = useState<Business[]>([])
   241→  const [loading, setLoading] = useState(true)
   242→  const [generating, setGenerating] = useState<string | null>(null)
   243→  const [selectedBusiness, setSelectedBusiness] = useState<Business | null>(null)
   244→  const [proposal, setProposal] = useState<ProposalData | null>(null)
   245→  const [proposalOpen, setProposalOpen] = useState(false)
   246→  const [loadingProposal, setLoadingProposal] = useState(false)
   247→  const [searchTerm, setSearchTerm] = useState('')
   248→  const [categoryFilter, setCategoryFilter] = useState('All')
   249→  const [page, setPage] = useState(1)
   250→  const pageSize = 8
   251→
   252→  const categories = ['All', 'Restaurant', 'Salon', 'Beauty Parlour', 'Spa', 'Gym', 'Clinic', 'Hotel', 'Real Estate', 'Dentist', 'Lawyer', 'School', 'Other']
   253→
   254→  // Fetch businesses
   255→  useEffect(() => {
   256→    const fetchBusinesses = async () => {
   257→      try {
   258→        const res = await fetch('/api/businesses?limit=200')
   259→        if (res.ok) {
   260→          const data = await res.json()
   261→          setBusinesses(data.businesses || [])
   262→        }
   263→      } catch { setBusinesses([]) }
   264→      finally { setLoading(false) }
   265→    }
   266→    fetchBusinesses()
   267→  }, [user?.id])
   268→
   269→  const filteredBusinesses = businesses.filter(b => {
   270→    const matchesSearch = !searchTerm || b.name.toLowerCase().includes(searchTerm.toLowerCase()) || b.city?.toLowerCase().includes(searchTerm.toLowerCase())
   271→    const matchesCategory = categoryFilter === 'All' || b.category === categoryFilter
   272→    return matchesSearch && matchesCategory
   273→  })
   274→
   275→  const totalPages = Math.ceil(filteredBusinesses.length / pageSize)
   276→  const paginatedBusinesses = filteredBusinesses.slice((page - 1) * pageSize, page * pageSize)
   277→
   278→  const proposalsGenerated = businesses.filter(b => b.proposalData).length
   279→
   280→  // Generate proposal
   281→  const handleGenerateProposal = async (business: Business, useAI = false) => {
   282→    setGenerating(business.id)
   283→    try {
   284→      const res = await fetch('/api/businesses/proposal', {
   285→        method: 'POST',
   286→        headers: { 'Content-Type': 'application/json' },
   287→        body: JSON.stringify({ businessId: business.id, useAI }),
   288→      })
   289→      if (res.ok) {
   290→        const data = await res.json()
   291→        setProposal(data.proposal)
   292→        setSelectedBusiness(business)
   293→        setProposalOpen(true)
   294→        toast({ title: 'Proposal Generated!', description: `Proposal for ${business.name} is ready.` })
   295→        // Refresh businesses list
   296→        const bizRes = await fetch('/api/businesses?limit=200')
   297→        if (bizRes.ok) {
   298→          const bizData = await bizRes.json()
   299→          setBusinesses(bizData.businesses || [])
   300→        }
   301→      } else {
   302→        toast({ title: 'Failed to generate proposal', variant: 'destructive' })
   303→      }
   304→    } catch {
   305→      toast({ title: 'Error generating proposal', variant: 'destructive' })
   306→    } finally {
   307→      setGenerating(null)
   308→    }
   309→  }
   310→
   311→  // View existing proposal
   312→  const handleViewProposal = async (business: Business) => {
   313→    setSelectedBusiness(business)
   314→    setLoadingProposal(true)
   315→    setProposalOpen(true)
   316→    try {
   317→      const res = await fetch(`/api/businesses/proposal?businessId=${business.id}`)
   318→      if (res.ok) {
   319→        const data = await res.json()
   320→        setProposal(data.proposal)
   321→      } else {
   322→        toast({ title: 'Failed to load proposal', variant: 'destructive' })
   323→      }
   324→    } catch {
   325→      toast({ title: 'Error loading proposal', variant: 'destructive' })
   326→    } finally {
   327→      setLoadingProposal(false)
   328→    }
   329→  }
   330→
   331→  // Export PDF
   332→  const handleExportPDF = async (businessId: string) => {
   333→    try {
   334→      toast({ title: 'Generating PDF...', description: 'Your proposal PDF is being prepared.' })
   335→      const res = await fetch(`/api/businesses/proposal/pdf?businessId=${businessId}`)
   336→      if (res.ok) {
   337→        const html = await res.text()
   338→        // Open in new window for printing
   339→        const printWindow = window.open('', '_blank')
   340→        if (printWindow) {
   341→          printWindow.document.write(html)
   342→          printWindow.document.close()
   343→          setTimeout(() => {
   344→            printWindow.print()
   345→          }, 500)
   346→        }
   347→        toast({ title: 'PDF Ready!', description: 'Use the print dialog to save as PDF.' })
   348→      } else {
   349→        toast({ title: 'Failed to generate PDF', variant: 'destructive' })
   350→      }
   351→    } catch {
   352→      toast({ title: 'Error exporting PDF', variant: 'destructive' })
   353→    }
   354→  }
   355→
   356→  const formatPrice = (price: number) => {
   357→    return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 0 }).format(price)
   358→  }
   359→
   360→  if (loading) {
   361→    return (
   362→      <div className="flex items-center justify-center h-64">
   363→        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-amber-500" />
   364→      </div>
   365→    )
   366→  }
   367→
   368→  return (
   369→    <div className="space-y-6">
   370→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   371→        <div>
   372→          <h1 className="text-2xl font-bold text-slate-900 flex items-center gap-2">
   373→            <FileText className="h-6 w-6 text-amber-500" />
   374→            AI Website Proposal
   375→          </h1>
   376→          <p className="text-muted-foreground">Generate professional 3-tier proposals for businesses</p>
   377→        </div>
   378→      </div>
   379→
   380→      {/* Stats */}
   381→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   382→        <Card className="border-0 shadow-sm">
   383→          <CardContent className="p-4">
   384→            <div className="flex items-center gap-3">
   385→              <div className="h-10 w-10 rounded-xl bg-slate-50 flex items-center justify-center">
   386→                <Building2 className="h-5 w-5 text-slate-600" />
   387→              </div>
   388→              <div>
   389→                <p className="text-2xl font-bold text-slate-900">{businesses.length}</p>
   390→                <p className="text-xs text-muted-foreground">Businesses</p>
   391→              </div>
   392→            </div>
   393→          </CardContent>
   394→        </Card>
   395→        <Card className="border-0 shadow-sm">
   396→          <CardContent className="p-4">
   397→            <div className="flex items-center gap-3">
   398→              <div className="h-10 w-10 rounded-xl bg-emerald-50 flex items-center justify-center">
   399→                <FileText className="h-5 w-5 text-emerald-600" />
   400→              </div>
   401→              <div>
   402→                <p className="text-2xl font-bold text-emerald-600">{proposalsGenerated}</p>
   403→                <p className="text-xs text-muted-foreground">Proposals</p>
   404→              </div>
   405→            </div>
   406→          </CardContent>
   407→        </Card>
   408→        <Card className="border-0 shadow-sm">
   409→          <CardContent className="p-4">
   410→            <div className="flex items-center gap-3">
   411→              <div className="h-10 w-10 rounded-xl bg-amber-50 flex items-center justify-center">
   412→                <Target className="h-5 w-5 text-amber-600" />
   413→              </div>
   414→              <div>
   415→                <p className="text-2xl font-bold text-amber-600">{businesses.filter(b => !b.hasWebsite).length}</p>
   416→                <p className="text-xs text-muted-foreground">No Website</p>
   417→              </div>
   418→            </div>
   419→          </CardContent>
   420→        </Card>
   421→        <Card className="border-0 shadow-sm">
   422→          <CardContent className="p-4">
   423→            <div className="flex items-center gap-3">
   424→              <div className="h-10 w-10 rounded-xl bg-red-50 flex items-center justify-center">
   425→                <Package className="h-5 w-5 text-red-600" />
   426→              </div>
   427→              <div>
   428→                <p className="text-2xl font-bold text-red-600">{businesses.filter(b => !b.hasWebsite && !b.proposalData).length}</p>
   429→                <p className="text-xs text-muted-foreground">Need Proposal</p>
   430→              </div>
   431→            </div>
   432→          </CardContent>
   433→        </Card>
   434→      </div>
   435→
   436→      {/* Filters */}
   437→      <Card className="border-0 shadow-sm">
   438→        <CardContent className="p-4">
   439→          <div className="flex flex-col sm:flex-row gap-4">
   440→            <div className="flex-1 relative">
   441→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   442→              <Input
   443→                placeholder="Search by name or city..."
   444→                value={searchTerm}
   445→                onChange={(e) => { setSearchTerm(e.target.value); setPage(1) }}
   446→                className="pl-10 h-10"
   447→              />
   448→            </div>
   449→            <Select value={categoryFilter} onValueChange={(v) => { setCategoryFilter(v); setPage(1) }}>
   450→              <SelectTrigger className="w-full sm:w-48 h-10">
   451→                <SelectValue placeholder="Category" />
   452→              </SelectTrigger>
   453→              <SelectContent>
   454→                {categories.map((cat) => (
   455→                  <SelectItem key={cat} value={cat}>{cat}</SelectItem>
   456→                ))}
   457→              </SelectContent>
   458→            </Select>
   459→          </div>
   460→        </CardContent>
   461→      </Card>
   462→
   463→      {/* Businesses Table */}
   464→      <Card className="border-0 shadow-sm">
   465→        <CardContent className="p-0">
   466→          <ScrollArea className="w-full">
   467→            <div className="min-w-[700px]">
   468→              <Table>
   469→                <TableHeader>
   470→                  <TableRow className="bg-slate-50/50">
   471→                    <TableHead>Business</TableHead>
   472→                    <TableHead>Category</TableHead>
   473→                    <TableHead>Location</TableHead>
   474→                    <TableHead>Website</TableHead>
   475→                    <TableHead>Proposal</TableHead>
   476→                    <TableHead className="text-right">Actions</TableHead>
   477→                  </TableRow>
   478→                </TableHeader>
   479→                <TableBody>
   480→                  <AnimatePresence>
   481→                    {paginatedBusinesses.map((business) => (
   482→                      <motion.tr
   483→                        key={business.id}
   484→                        initial={{ opacity: 0 }}
   485→                        animate={{ opacity: 1 }}
   486→                        exit={{ opacity: 0 }}
   487→                        className="hover:bg-slate-50 cursor-pointer"
   488→                        onClick={() => business.proposalData ? handleViewProposal(business) : handleGenerateProposal(business)}
   489→                      >
   490→                        <TableCell>
   491→                          <div className="flex items-center gap-2">
   492→                            <div className={`h-8 w-8 rounded-lg flex items-center justify-center shrink-0 ${business.proposalData ? 'bg-emerald-50' : business.hasWebsite ? 'bg-slate-50' : 'bg-amber-50'}`}>
   493→                              <Building2 className={`h-4 w-4 ${business.proposalData ? 'text-emerald-600' : 'text-amber-600'}`} />
   494→                            </div>
   495→                            <div>
   496→                              <span className="font-medium text-sm block max-w-[160px] truncate">{business.name}</span>
   497→                              {business.leadScore != null && (
   498→                                <span className="text-[10px] text-muted-foreground">Lead: {business.leadScore}/100</span>
   499→                              )}
   500→                            </div>
   501→                          </div>
   502→                        </TableCell>
   503→                        <TableCell>
   504→                          <Badge variant="secondary" className="text-xs">{business.category}</Badge>
   505→                        </TableCell>
   506→                        <TableCell className="text-sm text-muted-foreground">
   507→                          {[business.city, business.country].filter(Boolean).join(', ') || '-'}
   508→                        </TableCell>
   509→                        <TableCell>
   510→                          {business.hasWebsite ? (
   511→                            <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100 text-[10px] border border-emerald-200">Yes</Badge>
   512→                          ) : (
   513→                            <Badge className="bg-red-100 text-red-800 hover:bg-red-100 text-[10px] border border-red-200">No</Badge>
   514→                          )}
   515→                        </TableCell>
   516→                        <TableCell>
   517→                          {business.proposalData ? (
   518→                            <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100 text-[10px] border border-emerald-200">
   519→                              <CheckCircle2 className="h-3 w-3 mr-1" />Generated
   520→                            </Badge>
   521→                          ) : (
   522→                            <Badge variant="outline" className="text-[10px]">Pending</Badge>
   523→                          )}
   524→                        </TableCell>
   525→                        <TableCell className="text-right">
   526→                          <div className="flex items-center justify-end gap-1">
   527→                            {business.proposalData ? (
   528→                              <>
   529→                                <Button size="sm" variant="ghost" className="h-7 text-xs text-amber-700 hover:bg-amber-50" onClick={(e) => { e.stopPropagation(); handleViewProposal(business) }}>
   530→                                  <Eye className="h-3.5 w-3.5 mr-1" />View
   531→                                </Button>
   532→                                <Button size="sm" variant="ghost" className="h-7 text-xs text-slate-600 hover:bg-slate-50" onClick={(e) => { e.stopPropagation(); handleExportPDF(business.id) }}>
   533→                                  <Download className="h-3.5 w-3.5 mr-1" />PDF
   534→                                </Button>
   535→                              </>
