    61→    61→function formatRelativeTime(dateStr: string): string {
    62→    62→  const date = new Date(dateStr)
    63→    63→  const now = new Date()
    64→    64→  const diffMs = now.getTime() - date.getTime()
    65→    65→  const diffSeconds = Math.floor(diffMs / 1000)
    66→    66→  const diffMinutes = Math.floor(diffSeconds / 60)
    67→    67→  const diffHours = Math.floor(diffMinutes / 60)
    68→    68→  const diffDays = Math.floor(diffHours / 24)
    69→    69→
    70→    70→  if (diffSeconds < 60) return 'Just now'
    71→    71→  if (diffMinutes < 60) return `${diffMinutes}m ago`
    72→    72→  if (diffHours < 24) return `${diffHours}h ago`
    73→    73→  if (diffDays < 7) return `${diffDays}d ago`
    74→    74→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
    75→    75→}

   237→   237→export function NotificationsView() {
   238→   238→  const { user } = useAppStore()
   239→   239→  const { toast } = useToast()
   240→   240→
   241→   241→  // Notification state
   242→   242→  const [notifications, setNotifications] = useState<ApiNotification[]>([])
   243→   243→  const [apiUnreadCount, setApiUnreadCount] = useState<number>(0)
   244→   244→  const [loading, setLoading] = useState(true)
   245→   245→  const [preferencesLoading, setPreferencesLoading] = useState(true)
   246→   246→  const [error, setError] = useState<string | null>(null)
   247→   247→  const [activeTab, setActiveTab] = useState<string>('all')
   248→   248→  const [markingRead, setMarkingRead] = useState<string | null>(null)
   249→   249→
   250→   250→  // Preferences state (loaded from API)
   251→   251→  const [preferences, setPreferences] = useState<NotificationPreference[]>(DEFAULT_PREFERENCES)
   252→   252→
   253→   253→  // Fetch notifications
   254→   254→  const fetchNotifications = useCallback(async () => {
   255→   255→    if (!user?.id) return
   256→   256→    setLoading(true)
   257→   257→    setError(null)
   258→   258→    try {
   259→   259→      const res = await fetch('/api/user/notifications?limit=50', {
   260→   260→        credentials: 'include',
   261→   261→      })
   262→   262→      if (!res.ok) {
   263→   263→        const body = await res.json().catch(() => ({}))
   264→   264→        throw new Error(body.error || 'Failed to fetch notifications')
   265→   265→      }
   266→   266→      const json = await res.json()
   267→   267→      setNotifications(json.notifications ?? [])
   268→   268→      setApiUnreadCount(json.unreadCount ?? 0)
   269→   269→    } catch (err: any) {
   270→   270→      console.error('Failed to fetch notifications:', err)
   271→   271→      setError(err.message || 'Failed to load notifications')
   272→   272→      toast({
   273→   273→        title: 'Error',
   274→   274→        description: 'Failed to load notifications. Please try again.',
   275→   275→        variant: 'destructive',
   276→   276→      })
   277→   277→    } finally {
   278→   278→      setLoading(false)
   279→   279→    }
   280→   280→  }, [user?.id, toast])
   281→   281→
   282→   282→  // Fetch preferences from API
   283→   283→  const fetchPreferences = useCallback(async () => {
   284→   284→    if (!user?.id) return
   285→   285→    setPreferencesLoading(true)
   286→   286→    try {
   287→   287→      const res = await fetch('/api/user/notifications/preferences', {
   288→   288→        credentials: 'include',
   289→   289→      })
   290→   290→      if (!res.ok) {
   291→   291→        throw new Error('Failed to fetch preferences')
   292→   292→      }
   293→   293→      const json = await res.json()
   294→   294→      // Map API response to our local state structure
   295→   295→      const apiPrefs = json.preferences as Record<string, { id: string; itemKey: string; enabled: boolean }[]>
   296→   296→
   297→   297→      setPreferences((prev) =>
   298→   298→        prev.map((cat) => {
   299→   299→          const apiItems = apiPrefs[cat.category]
   300→   300→          if (!apiItems) return cat
   301→   301→          return {
   302→   302→            ...cat,
   303→   303→            items: cat.items.map((item) => {
   304→   304→              const apiItem = apiItems.find((a) => a.itemKey === item.itemKey)
   305→   305→              if (apiItem) {
   306→   306→                return { ...item, id: apiItem.id, enabled: apiItem.enabled }
   307→   307→              }
   308→   308→              return item
   309→   309→            }),
   310→   310→          }
   311→   311→        })
   312→   312→      )
   313→   313→    } catch (err) {
   314→   314→      console.error('Failed to fetch preferences:', err)
   315→   315→      // Keep defaults on error
   316→   316→    } finally {
   317→   317→      setPreferencesLoading(false)
   318→   318→    }
   319→   319→  }, [user?.id])
   320→   320→
   321→   321→  useEffect(() => {
   322→   322→    fetchNotifications()
   323→   323→    fetchPreferences()
   324→   324→  }, [fetchNotifications, fetchPreferences])
   325→   325→
   326→   326→  // Derived state
   327→   327→  const unreadCount = useMemo(() => notifications.filter((n) => !n.read).length, [notifications])
   328→   328→
   329→   329→  const filteredNotifications = useMemo(() => {
   330→   330→    switch (activeTab) {
   331→   331→      case 'unread':
   332→   332→        return notifications.filter((n) => !n.read)
   333→   333→      default:
   334→   334→        return notifications
   335→   335→    }
   336→   336→  }, [notifications, activeTab])
   337→   337→
   338→   338→  // Actions
   339→   339→  const markAsRead = async (id: string) => {
   340→   340→    if (!user?.id) return

   342→   342→    const prev = notifications
   343→   343→    setNotifications((ns) =>
   344→   344→      ns.map((n) => (n.id === id ? { ...n, read: true } : n))
   345→   345→    )
   346→   346→    setMarkingRead(id)
   347→   347→    try {
   348→   348→      const res = await fetch('/api/user/notifications', {
   349→   349→        method: 'PUT',
   350→   350→        headers: { 'Content-Type': 'application/json' },
   351→   351→        credentials: 'include',
   352→   352→        body: JSON.stringify({ notificationId: id }),
   353→   353→      })

   355→   355→        throw new Error('Failed to mark as read')
   356→   356→      }
   357→   357→    } catch (err) {
   358→   358→      console.error('Failed to mark as read:', err)
   359→   359→      // Rollback
   360→   360→      setNotifications(prev)
   361→   361→      toast({
   362→   362→        title: 'Error',
   363→   363→        description: 'Failed to mark notification as read.',
   364→   364→        variant: 'destructive',
   365→   365→      })
   366→   366→    } finally {
   367→   367→      setMarkingRead(null)
   368→   368→    }
   369→   369→  }

   376→   376→    try {
   377→   377→      const res = await fetch('/api/user/notifications', {

   390→   390→    } catch (err) {
   391→   391→      console.error('Failed to mark all as read:', err)
   392→   392→      // Rollback
   393→   393→      setNotifications(prev)
   394→   394→      toast({
   395→   395→        title: 'Error',
   396→   396→        description: 'Failed to mark all notifications as read.',

   466→   466→    } catch (err) {

   468→   468→      // Rollback

   470→   470→        prev.map((cat) =>
   471→   471→          cat.category === categoryId

   475→   475→                  i.itemKey === itemKey ? { ...i, enabled: !newEnabled, saving: false } : i

   491→   491→      prev.map((cat) =>
   492→   492→        cat.category === categoryId
   493→   493→          ? {
   494→   494→              ...cat,
   495→   495→              items: cat.items.map((i) =>
   496→   496→                i.itemKey === itemKey ? { ...i, saving: false } : i
