    57→function roleBadgeStyle(role: string): string {
    58→  switch (role) {
    59→    case 'super_admin':
    60→      return 'bg-red-100 text-red-700 border-red-200 dark:bg-red-500/20 dark:text-red-400 dark:border-red-500/30'
    61→    case 'admin':
    62→      return 'bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-500/20 dark:text-amber-400 dark:border-amber-500/30'
    63→    case 'agency_owner':
    64→      return 'bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-500/20 dark:text-emerald-400 dark:border-emerald-500/30'

    85→function formatDate(dateStr: string | null | undefined): string {
    86→  if (!dateStr) return 'N/A'
    87→  try {
    88→    return new Date(dateStr).toLocaleDateString('en-US', {
    89→      year: 'numeric',
    90→      month: 'long',
    91→      day: 'numeric',
    92→    })
    93→  } catch {
    94→    return dateStr
    95→  }
    96→}
