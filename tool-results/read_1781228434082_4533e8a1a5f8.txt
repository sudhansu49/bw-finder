   132→export function SearchView() {
   133→  const { user, setCurrentView } = useAppStore()
   134→  const { toast } = useToast()
   135→  const [country, setCountry] = useState('')
   136→  const [state, setState] = useState('')
   137→  const [city, setCity] = useState('')
   138→  const [category, setCategory] = useState('')
   139→  const [customCountry, setCustomCountry] = useState('')
   140→  const [customCategory, setCustomCategory] = useState('')
   141→  const [loading, setLoading] = useState(false)
   142→  const [results, setResults] = useState<Business[]>([])
   143→  const [showNoWebsiteOnly, setShowNoWebsiteOnly] = useState(false)
   144→  const [searched, setSearched] = useState(false)
   145→  const [searchJobInfo, setSearchJobInfo] = useState<SearchJobInfo | null>(null)
   146→  const [selectedBusiness, setSelectedBusiness] = useState<Business | null>(null)
   147→  const [detailOpen, setDetailOpen] = useState(false)
   148→  const [addedLeads, setAddedLeads] = useState<Set<string>>(new Set())
   149→  const [addingLead, setAddingLead] = useState<string | null>(null)
   150→
   151→  const effectiveCountry = country === 'Other' ? customCountry : country
   152→  const effectiveCategory = category === 'Other' ? customCategory : category
   153→
   154→  const handleSearch = async () => {
   155→    if (!effectiveCountry && !customCountry) {
   156→      toast({ title: 'Please select or enter a country', variant: 'destructive' })
   157→      return
   158→    }
   159→    if (!effectiveCategory && !customCategory) {
   160→      toast({ title: 'Please select or enter a business category', variant: 'destructive' })
   161→      return
   162→    }
   163→
   164→    setLoading(true)
   165→    setSearched(true)
   166→    setSearchJobInfo(null)
   167→
   168→    try {
   169→      const res = await fetch('/api/businesses/search', {
   170→        method: 'POST',
   171→        headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({
   173→          country: effectiveCountry,
   174→          state: state.trim(),
   175→          city: city.trim(),
   176→          category: effectiveCategory,
   177→          userId: user?.id || 'demo',
   178→        }),
   179→      })
   180→
   181→      if (res.ok) {
   182→        const data = await res.json()
   183→        setResults(data.businesses || [])
   184→        setSearchJobInfo(data.searchJob || null)
   185→        toast({
   186→          title: 'Search Complete!',
   187→          description: `Found ${data.businesses?.length || 0} businesses. ${data.searchJob?.duplicatesFound || 0} duplicates were merged.`,
   188→        })
   189→      } else {
   190→        toast({
   191→          title: 'Search Failed',
   192→          description: 'Could not complete the search. Please try again.',
   193→          variant: 'destructive',
   194→        })
   195→        setResults([])
   196→      }
   197→    } catch {
   198→      toast({
   199→        title: 'Search Failed',
   200→        description: 'Network error. Please try again.',
   201→        variant: 'destructive',
   202→      })
   203→      setResults([])
   204→    } finally {
   205→      setLoading(false)
   206→    }
   207→  }
   208→
   209→  const handleAddAsLead = async (business: Business) => {
   210→    if (!business.id || addedLeads.has(business.id)) return
   211→
   212→    setAddingLead(business.id || business.name)
   213→    try {
   214→      let businessId = business.id
   215→
   216→      if (!businessId) {
   217→        const businessRes = await fetch('/api/businesses', {
   218→          method: 'POST',
   219→          headers: { 'Content-Type': 'application/json' },
   220→          body: JSON.stringify({
   221→            name: business.name,
   222→            category: business.category,
   223→            address: business.address || '',
   224→            city: business.city || '',
   225→            state: business.state || '',
   226→            country: business.country || '',
   227→            phone: business.phone || '',
   228→            email: business.email || '',
   229→            website: business.website || null,
   230→            hasWebsite: business.hasWebsite,
   231→            facebookUrl: business.facebookUrl || null,
   232→            instagramUrl: business.instagramUrl || null,
   233→            linkedinUrl: business.linkedinUrl || null,
   234→          }),
   235→        })
   236→        if (businessRes.ok) {
   237→          const businessData = await businessRes.json()
   238→          businessId = businessData.business?.id || businessData.id
   239→        }
   240→      }
   241→
   242→      if (businessId) {
   243→        await fetch('/api/leads', {
   244→          method: 'POST',
   245→          headers: { 'Content-Type': 'application/json' },
   246→          body: JSON.stringify({
   247→            businessId,
   248→            userId: user?.id || 'demo',
   249→            status: 'New',
   250→            priority: business.hasWebsite ? 'low' : 'medium',
   251→            estimatedValue: business.hasWebsite ? 800 : 1500,
   252→            notes: `Lead from discovery search: ${business.name} in ${business.city || ''}, ${business.country || ''}`,
   253→          }),
   254→        })
   255→      }
   256→
   257→      if (business.id) {
   258→        setAddedLeads(prev => new Set([...prev, business.id!]))
   259→      }
   260→
   261→      toast({
   262→        title: 'Lead Added!',
   263→        description: `${business.name} has been added to your leads pipeline.`,
   264→      })
   265→    } catch {
   266→      toast({
   267→        title: 'Error',
   268→        description: `Failed to add ${business.name} as lead.`,
   269→        variant: 'destructive',
   270→      })
   271→    } finally {
   272→      setAddingLead(null)
   273→    }
   274→  }
   275→
   276→  const handleAddAllNoWebsiteAsLeads = async () => {
   277→    const noWebsiteBusinesses = filteredResults.filter(b => !b.hasWebsite && b.id && !addedLeads.has(b.id))
   278→    let added = 0
   279→
   280→    for (const business of noWebsiteBusinesses) {
   281→      try {
   282→        await fetch('/api/leads', {
   283→          method: 'POST',
   284→          headers: { 'Content-Type': 'application/json' },
   285→          body: JSON.stringify({
   286→            businessId: business.id,
   287→            userId: user?.id || 'demo',
   288→            status: 'New',
   289→            priority: 'medium',
   290→            estimatedValue: 1500,
   291→            notes: `Bulk lead from discovery: ${business.name}`,
   292→          }),
   293→        })
   294→        if (business.id) {
   295→          setAddedLeads(prev => new Set([...prev, business.id!]))
   296→        }
   297→        added++
   298→      } catch {
   299→        // Skip failed ones
   300→      }
   301→    }
   302→
   303→    toast({
   304→      title: `${added} Leads Added!`,
   305→      description: `Added ${added} businesses without websites to your leads pipeline.`,
   306→    })
   307→  }
   308→
   309→  // Website status badge component
   310→  const WebsiteStatusBadge = ({ status, hasWebsite }: { status?: string | null, hasWebsite: boolean }) => {
   311→    if (status === 'HAS_WEBSITE') {
   312→      return (
   313→        <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100 text-xs font-semibold border border-emerald-200">
   314→          <Globe className="h-3 w-3 mr-1" />
   315→          Website Exists
   316→        </Badge>
   317→      )
   318→    }
   319→    if (status === 'SOCIAL_ONLY') {
   320→      return (
   321→        <Badge className="bg-red-100 text-red-800 hover:bg-red-100 text-xs font-semibold border border-red-200">
   322→          <Unplug className="h-3 w-3 mr-1" />
   323→          Social Only
   324→        </Badge>
   325→      )
   326→    }
   327→    // NO_WEBSITE or not detected yet
   328→    return (
   329→      <Badge className="bg-red-100 text-red-800 hover:bg-red-100 text-xs font-semibold border border-red-200">
   330→        <Unplug className="h-3 w-3 mr-1" />
   331→        No Website
   332→      </Badge>
   333→    )
   334→  }
   335→
   336→  // Lead score color
   337→  const getScoreColor = (score: number | null | undefined) => {
   338→    if (!score) return 'text-slate-400'
   339→    if (score >= 70) return 'text-emerald-600'
   340→    if (score >= 40) return 'text-amber-600'
   341→    return 'text-red-500'
   342→  }
   343→
   344→  const getScoreBg = (score: number | null | undefined) => {
   345→    if (!score) return 'bg-slate-50'
   346→    if (score >= 70) return 'bg-emerald-50'
   347→    if (score >= 40) return 'bg-amber-50'
   348→    return 'bg-red-50'
   349→  }
   350→
   351→  const filteredResults = showNoWebsiteOnly
   352→    ? results.filter((b) => b.websiteStatus !== 'HAS_WEBSITE')
   353→    : results
   354→
   355→  const noWebsiteCount = results.filter((b) => !b.hasWebsite).length
   356→  const withWebsiteCount = results.filter((b) => b.hasWebsite).length
   357→
   358→  const openDetail = (business: Business) => {
   359→    setSelectedBusiness(business)
   360→    setDetailOpen(true)
   361→  }
   362→
   363→  return (
   364→    <div className="space-y-6">
   365→      <div>
   366→        <h1 className="text-2xl font-bold text-slate-900">Lead Discovery Engine</h1>
   367→        <p className="text-muted-foreground">Find local businesses without websites across any location</p>
   368→      </div>
   369→
   370→      {/* Search Form */}
   371→      <Card className="border-0 shadow-sm">
   372→        <CardHeader className="pb-3">
   373→          <CardTitle className="text-base flex items-center gap-2">
   374→            <Search className="h-4 w-4 text-amber-500" />
   375→            Search Criteria
   376→          </CardTitle>
   377→        </CardHeader>
   378→        <CardContent className="p-6 pt-0">
   379→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   380→            {/* Country */}
   381→            <div className="space-y-2">
   382→              <Label htmlFor="country" className="text-sm font-medium">
   383→                Country <span className="text-red-500">*</span>
   384→              </Label>
   385→              <Select value={country} onValueChange={setCountry}>
   386→                <SelectTrigger className="h-11">
   387→                  <SelectValue placeholder="Select country" />
   388→                </SelectTrigger>
   389→                <SelectContent>
   390→                  {countries.map((c) => (
   391→                    <SelectItem key={c} value={c}>
   392→                      {c}
   393→                    </SelectItem>
   394→                  ))}
   395→                </SelectContent>
   396→              </Select>
   397→              {country === 'Other' && (
   398→                <Input
   399→                  placeholder="Enter country name"
   400→                  value={customCountry}
   401→                  onChange={(e) => setCustomCountry(e.target.value)}
   402→                  className="h-10"
   403→                />
   404→              )}
   405→            </div>
   406→
   407→            {/* State */}
   408→            <div className="space-y-2">
   409→              <Label htmlFor="state" className="text-sm font-medium">State / Province</Label>
   410→              <div className="relative">
   411→                <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   412→                <Input
   413→                  id="state"
   414→                  placeholder="e.g. Maharashtra, California"
   415→                  value={state}
   416→                  onChange={(e) => setState(e.target.value)}
   417→                  className="pl-10 h-11"
   418→                />
   419→              </div>
   420→            </div>
   421→
   422→            {/* City */}
   423→            <div className="space-y-2">
   424→              <Label htmlFor="city" className="text-sm font-medium">City</Label>
   425→              <div className="relative">
   426→                <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   427→                <Input
   428→                  id="city"
   429→                  placeholder="e.g. Mumbai, New York"
   430→                  value={city}
   431→                  onChange={(e) => setCity(e.target.value)}
   432→                  className="pl-10 h-11"
   433→                />
   434→              </div>
   435→            </div>
   436→
   437→            {/* Category */}
   438→            <div className="space-y-2">
   439→              <Label className="text-sm font-medium">
   440→                Business Category <span className="text-red-500">*</span>
   441→              </Label>
   442→              <Select value={category} onValueChange={setCategory}>
   443→                <SelectTrigger className="h-11">
   444→                  <SelectValue placeholder="Select category" />
   445→                </SelectTrigger>
   446→                <SelectContent>
   447→                  {categories.map((cat) => (
   448→                    <SelectItem key={cat} value={cat}>
   449→                      {cat}
   450→                    </SelectItem>
   451→                  ))}
   452→                </SelectContent>
   453→              </Select>
   454→              {category === 'Other' && (
   455→                <Input
   456→                  placeholder="Enter business category"
   457→                  value={customCategory}
   458→                  onChange={(e) => setCustomCategory(e.target.value)}
   459→                  className="h-10"
   460→                />
   461→              )}
   462→            </div>
   463→          </div>
   464→
   465→          {/* Search Button */}
   466→          <div className="mt-5 flex items-center gap-3">
   467→            <Button
   468→              onClick={handleSearch}
   469→              disabled={loading || !effectiveCountry || !effectiveCategory}
   470→              className="h-11 px-8 bg-amber-500 hover:bg-amber-600 text-white font-semibold"
   471→            >
   472→              {loading ? (
   473→                <>
   474→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   475→                  Discovering Businesses...
   476→                </>
   477→              ) : (
   478→                <>
   479→                  <Search className="mr-2 h-4 w-4" />
   480→                  Discover Businesses
   481→                </>
   482→              )}
   483→            </Button>
   484→            {searched && !loading && (
   485→              <Button
   486→                variant="outline"
   487→                onClick={() => {
   488→                  setSearched(false)
   489→                  setResults([])
   490→                  setSearchJobInfo(null)
   491→                  setAddedLeads(new Set())
   492→                }}
   493→                className="h-11"
   494→              >
   495→                Clear Results
   496→              </Button>
   497→            )}
   498→          </div>
   499→        </CardContent>
   500→      </Card>
   501→
   502→      {/* Loading State */}
   503→      {loading && (
   504→        <motion.div
   505→          initial={{ opacity: 0 }}
   506→          animate={{ opacity: 1 }}
   507→          className="flex flex-col items-center justify-center py-16"
   508→        >
   509→          <div className="relative">
   510→            <div className="animate-spin rounded-full h-16 w-16 border-4 border-amber-200 border-t-amber-500" />
   511→            <Search className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 h-6 w-6 text-amber-500" />
   512→          </div>
   513→          <p className="mt-4 text-muted-foreground font-medium">Discovering businesses...</p>
   514→          <p className="text-sm text-muted-foreground mt-1">Searching multiple directories and business listings</p>
   515→          <div className="mt-4 flex items-center gap-2 text-xs text-muted-foreground">
   516→            <div className="flex items-center gap-1">
   517→              <div className="h-2 w-2 rounded-full bg-amber-500 animate-pulse" />
   518→              Web Search
   519→            </div>
   520→            <div className="flex items-center gap-1">
   521→              <div className="h-2 w-2 rounded-full bg-emerald-500 animate-pulse" />
   522→              Directory Scan
   523→            </div>
   524→            <div className="flex items-center gap-1">
   525→              <div className="h-2 w-2 rounded-full bg-blue-500 animate-pulse" />
   526→              Data Extraction
   527→            </div>
   528→          </div>
   529→        </motion.div>
   530→      )}
   531→
   532→      {/* Results */}
   533→      {!loading && searched && (
   534→        <div className="space-y-4">
   535→          {/* Results Header & Stats */}
   536→          <div className="flex flex-col gap-4">
   537→            {/* Stats Row */}
   538→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   539→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   540→                <p className="text-2xl font-bold text-slate-900">{results.length}</p>
   541→                <p className="text-xs text-muted-foreground">Total Found</p>
   542→              </div>
   543→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   544→                <p className="text-2xl font-bold text-red-600">{noWebsiteCount}</p>
   545→                <p className="text-xs text-muted-foreground">No Website</p>
   546→              </div>
   547→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   548→                <p className="text-2xl font-bold text-emerald-600">{withWebsiteCount}</p>
   549→                <p className="text-xs text-muted-foreground">Has Website</p>
   550→              </div>
   551→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   552→                <p className="text-2xl font-bold text-amber-600">
   553→                  {searchJobInfo?.duplicatesFound || 0}
   554→                </p>
   555→                <p className="text-xs text-muted-foreground">Duplicates Merged</p>
   556→              </div>
   557→            </div>
   558→
   559→            {/* Search Job Info */}
   560→            {searchJobInfo && searchJobInfo.sourcesUsed && (
   561→              <div className="flex items-center gap-2 text-xs text-muted-foreground bg-amber-50 rounded-lg px-4 py-2">
   562→                <AlertCircle className="h-3.5 w-3.5 text-amber-600" />
   563→                <span>Sources searched: <span className="font-medium text-amber-700">{searchJobInfo.sourcesUsed}</span></span>
   564→              </div>
   565→            )}
   566→
   567→            {/* Actions Row */}
   568→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   569→              <div className="flex items-center gap-3">
   570→                <h2 className="text-lg font-semibold">
   571→                  {filteredResults.length} result{filteredResults.length !== 1 ? 's' : ''}
   572→                </h2>
   573→                <div className="flex items-center gap-2">
   574→                  <Button
   575→                    variant={showNoWebsiteOnly ? 'default' : 'outline'}
   576→                    size="sm"
   577→                    onClick={() => setShowNoWebsiteOnly(!showNoWebsiteOnly)}
   578→                    className={showNoWebsiteOnly ? 'bg-amber-500 hover:bg-amber-600 text-white' : ''}
   579→                  >
   580→                    <Filter className="mr-1 h-3.5 w-3.5" />
   581→                    {showNoWebsiteOnly ? 'No Website Only' : 'Filter: No Website'}
   582→                  </Button>
   583→                </div>
   584→              </div>
   585→              <div className="flex items-center gap-2">
   586→                {noWebsiteCount > 0 && !showNoWebsiteOnly && (
   587→                  <Button
   588→                    size="sm"
   589→                    variant="outline"
   590→                    onClick={handleAddAllNoWebsiteAsLeads}
   591→                    className="border-amber-200 text-amber-700 hover:bg-amber-50"
   592→                  >
   593→                    <Plus className="mr-1 h-3.5 w-3.5" />
   594→                    Add All No-Website as Leads ({noWebsiteCount})
   595→                  </Button>
   596→                )}
   597→              </div>
   598→            </div>
   599→          </div>
   600→
   601→          {/* Results Table */}
   602→          <Card className="border-0 shadow-sm">
   603→            <CardContent className="p-0">
   604→              <ScrollArea className="w-full">
   605→                <div className="min-w-[900px]">
   606→                  <Table>
