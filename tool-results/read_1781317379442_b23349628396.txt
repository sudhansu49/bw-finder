   169→function LeadCard({ lead, onClick }: { lead: Lead; onClick: () => void }) {
   170→  const priority = PRIORITY_CONFIG[lead.priority] || PRIORITY_CONFIG.medium
   171→  const hasTasks = lead.leadTasks && lead.leadTasks.length > 0
   172→  const hasReminders = lead.reminders && lead.reminders.length > 0
   173→
   174→  return (
   175→    <motion.div
   176→      layout
   177→      initial={{ opacity: 0, scale: 0.95 }}
   178→      animate={{ opacity: 1, scale: 1 }}
   179→      className="group cursor-pointer"
   180→      onClick={onClick}
   181→    >
   182→      <Card className="border border-slate-200 hover:border-slate-300 hover:shadow-md transition-all duration-200 bg-white">
   183→        <CardContent className="p-3 space-y-2">
   184→          {/* Header */}
   185→          <div className="flex items-start justify-between gap-2">
   186→            <h4 className="text-sm font-semibold text-slate-900 leading-tight line-clamp-2">
   187→              {lead.business.name}
   188→            </h4>
   189→            <Badge variant="secondary" className={`text-[10px] shrink-0 ${priority.bg} ${priority.color} border-0`}>
   190→              {priority.label}
   191→            </Badge>
   192→          </div>
   193→
   194→          {/* Category & Location */}
   195→          <div className="flex items-center gap-2 flex-wrap">
   196→            <Badge variant="outline" className="text-[10px] h-5 px-1.5">
   197→              <Tag className="h-2.5 w-2.5 mr-0.5" />
   198→              {lead.business.category}
   199→            </Badge>
   200→            {lead.business.city && (
   201→              <span className="text-[10px] text-muted-foreground flex items-center gap-0.5">
   202→                <MapPin className="h-2.5 w-2.5" />
   203→                {lead.business.city}
   204→              </span>
   205→            )}
   206→          </div>
   207→
   208→          {/* Value */}
   209→          {lead.estimatedValue && (
   210→            <div className="flex items-center gap-1">
   211→              <DollarSign className="h-3 w-3 text-emerald-600" />
   212→              <span className="text-xs font-semibold text-emerald-700">
   213→                ₹{lead.estimatedValue.toLocaleString('en-IN')}
   214→              </span>
   215→            </div>
   216→          )}
   217→
   218→          {/* Indicators */}
   219→          <div className="flex items-center gap-2 pt-1 border-t border-slate-100">
   220→            {lead.business.leadScore != null && (
   221→              <span className="text-[10px] text-muted-foreground">
   222→                Score: <span className="font-semibold text-slate-700">{lead.business.leadScore}</span>
   223→              </span>
   224→            )}
   225→            {hasTasks && (
   226→              <span className="text-[10px] text-blue-600 flex items-center gap-0.5">
   227→                <ListTodo className="h-2.5 w-2.5" />
   228→                {lead.leadTasks.length}
   229→              </span>
   230→            )}
   231→            {hasReminders && (
   232→              <span className="text-[10px] text-amber-600 flex items-center gap-0.5">
   233→                <Bell className="h-2.5 w-2.5" />
   234→                {lead.reminders.length}
   235→              </span>
   236→            )}
   237→            <span className="ml-auto text-[10px] text-muted-foreground flex items-center gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   238→              Open <ChevronRight className="h-2.5 w-2.5" />
   239→            </span>
   240→          </div>
   241→        </CardContent>
   242→      </Card>
   243→    </motion.div>
   244→  )
   245→}

   247→// ── Pipeline Column ──────────────────────────────────────────────
   248→
   249→function PipelineColumn({
   250→  stage,
   251→  leads,
   252→  onLeadClick,
   253→}: {
   254→  stage: PipelineStage
   255→  leads: Lead[]
   256→  onLeadClick: (lead: Lead) => void
   257→}) {
   258→  const Icon = STAGE_ICONS[stage.id] || Circle
   259→  const totalValue = leads.reduce((sum, l) => sum + (l.estimatedValue || 0), 0)
   260→
   261→  return (
   262→    <div className="flex flex-col min-w-[280px] w-[280px]">
   263→      {/* Column Header */}
   264→      <div className="flex items-center gap-2 px-3 py-2 mb-2 rounded-lg bg-white border border-slate-200">
   265→        <div className="h-7 w-7 rounded-lg flex items-center justify-center" style={{ backgroundColor: stage.color + '20' }}>
   266→          <Icon className="h-4 w-4" style={{ color: stage.color }} />
   267→        </div>
   268→        <div className="flex-1 min-w-0">
   269→          <h3 className="text-xs font-bold text-slate-900 truncate">{stage.label}</h3>
   270→          <p className="text-[10px] text-muted-foreground">
   271→            {leads.length} lead{leads.length !== 1 ? 's' : ''} · ₹{totalValue.toLocaleString('en-IN')}
   272→          </p>
   273→        </div>
   274→        <Badge variant="secondary" className="h-5 text-[10px] px-1.5" style={{ backgroundColor: stage.color + '15', color: stage.color }}>
   275→          {leads.length}
   276→        </Badge>
   277→      </div>
   278→
   279→      {/* Lead Cards */}
   280→      <div className="flex-1 space-y-2 overflow-y-auto max-h-[calc(100vh-320px)] pr-1 custom-scrollbar">
   281→        <AnimatePresence>
   282→          {leads.map((lead) => (
   283→            <LeadCard key={lead.id} lead={lead} onClick={() => onLeadClick(lead)} />
   284→          ))}
   285→        </AnimatePresence>
   286→        {leads.length === 0 && (
   287→          <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   288→            <Icon className="h-8 w-8 mb-2 opacity-20" style={{ color: stage.color }} />
   289→            <p className="text-xs">No leads here yet</p>
   290→          </div>
   291→        )}
   292→      </div>
   293→    </div>
   294→  )
   295→}

   299→function LeadDetailPanel({
   300→  lead,
   301→  onClose,
   302→  onRefresh,
   303→}: {
   304→  lead: Lead
   305→  onClose: () => void
   306→  onRefresh: () => void
   307→}) {
   308→  const { user } = useAppStore()
   309→  const { toast } = useToast()
   310→  const [activeTab, setActiveTab] = useState('details')
   311→  const [notes, setNotes] = useState<LeadNote[]>([])
   312→  const [tasks, setTasks] = useState<LeadTask[]>([])
   313→  const [reminders, setReminders] = useState<Reminder[]>([])
   314→  const [activities, setActivities] = useState<ActivityLogItem[]>([])
   315→  const [loadingDetails, setLoadingDetails] = useState(true)
   316→
   317→  // Form states
   318→  const [newNote, setNewNote] = useState('')
   319→  const [newTask, setNewTask] = useState('')
   320→  const [newTaskDue, setNewTaskDue] = useState('')
   321→  const [newReminder, setNewReminder] = useState('')
   322→  const [newReminderDue, setNewReminderDue] = useState('')
   323→  const [selectedStatus, setSelectedStatus] = useState(lead.status)
   324→  const [selectedPriority, setSelectedPriority] = useState(lead.priority)
   325→  const [estimatedValue, setEstimatedValue] = useState(lead.estimatedValue?.toString() || '')
   326→
   327→  // Load lead details
   328→  useEffect(() => {
   329→    const loadDetails = async () => {
   330→      setLoadingDetails(true)
   331→      try {
   332→        const [notesRes, tasksRes, remindersRes, activitiesRes] = await Promise.all([
   333→          fetch(`/api/crm/notes?leadId=${lead.id}`),
   334→          fetch(`/api/crm/tasks?leadId=${lead.id}`),
   335→          fetch(`/api/crm/reminders?leadId=${lead.id}`),
   336→          fetch(`/api/crm/activities?leadId=${lead.id}`),
   337→        ])
   338→        if (notesRes.ok) setNotes((await notesRes.json()).notes || [])
   339→        if (tasksRes.ok) setTasks((await tasksRes.json()).tasks || [])
   340→        if (remindersRes.ok) setReminders((await remindersRes.json()).reminders || [])
   341→        if (activitiesRes.ok) setActivities((await activitiesRes.json()).activities || [])
   342→      } catch {
   343→        // Silent fail
   344→      } finally {
   345→        setLoadingDetails(false)
   346→      }
   347→    }
   348→    loadDetails()
   349→  }, [lead.id])
   350→
   351→  // Update lead
   352→  const handleUpdateLead = async () => {
   353→    try {
   354→      const res = await fetch('/api/crm/pipeline', {
   355→        method: 'PATCH',
   356→        headers: { 'Content-Type': 'application/json' },
   357→        body: JSON.stringify({
   358→          leadId: lead.id,
   359→          status: selectedStatus,
   360→          priority: selectedPriority,
   361→          estimatedValue: estimatedValue ? parseFloat(estimatedValue) : null,
   362→          userId: user?.id || 'demo',
   363→        }),
   364→      })
   365→      if (res.ok) {
   366→        toast({ title: 'Lead updated!' })
   367→        onRefresh()
   368→      }
   369→    } catch {
   370→      toast({ title: 'Failed to update lead', variant: 'destructive' })
   371→    }
   372→  }
   373→
   374→  // Add note
   375→  const handleAddNote = async () => {
   376→    if (!newNote.trim()) return
   377→    try {
   378→      const res = await fetch('/api/crm/notes', {
   379→        method: 'POST',
   380→        headers: { 'Content-Type': 'application/json' },
   381→        body: JSON.stringify({ leadId: lead.id, userId: user?.id || 'demo', content: newNote.trim() }),
   382→      })
   383→      if (res.ok) {
   384→        const data = await res.json()
   385→        setNotes(prev => [data.note, ...prev])
   386→        setNewNote('')
   387→        toast({ title: 'Note added!' })
   388→      }
   389→    } catch {
   390→      toast({ title: 'Failed to add note', variant: 'destructive' })
   391→    }
   392→  }
   393→
   394→  // Add task
   395→  const handleAddTask = async () => {
   396→    if (!newTask.trim()) return
   397→    try {
   398→      const res = await fetch('/api/crm/tasks', {
   399→        method: 'POST',
   400→        headers: { 'Content-Type': 'application/json' },
   401→        body: JSON.stringify({
   402→          leadId: lead.id,
   403→          userId: user?.id || 'demo',
   404→          title: newTask.trim(),
   405→          dueDate: newTaskDue || null,
   406→        }),
   407→      })
   408→      if (res.ok) {
   409→        const data = await res.json()
   410→        setTasks(prev => [data.task, ...prev])
   411→        setNewTask('')
   412→        setNewTaskDue('')
   413→        toast({ title: 'Task added!' })
   414→      }
   415→    } catch {
   416→      toast({ title: 'Failed to add task', variant: 'destructive' })
   417→    }
   418→  }
   419→
   420→  // Toggle task
   421→  const handleToggleTask = async (taskId: string, completed: boolean) => {
   422→    try {
   423→      const res = await fetch('/api/crm/tasks', {
   424→        method: 'PATCH',
   425→        headers: { 'Content-Type': 'application/json' },
   426→        body: JSON.stringify({ taskId, completed: !completed }),
   427→      })
   428→      if (res.ok) {
   429→        setTasks(prev => prev.map(t => t.id === taskId ? { ...t, completed: !completed, completedAt: !completed ? new Date().toISOString() : null } : t))
   430→        toast({ title: completed ? 'Task reopened' : 'Task completed!' })
   431→      }
   432→    } catch {
   433→      toast({ title: 'Failed to update task', variant: 'destructive' })
   434→    }
   435→  }
   436→
   437→  // Delete task
   438→  const handleDeleteTask = async (taskId: string) => {
   439→    try {
   440→      const res = await fetch(`/api/crm/tasks?taskId=${taskId}`, { method: 'DELETE' })
   441→      if (res.ok) {
   442→        setTasks(prev => prev.filter(t => t.id !== taskId))
   443→        toast({ title: 'Task deleted' })
   444→      }
   445→    } catch {
   446→      toast({ title: 'Failed to delete task', variant: 'destructive' })
   447→    }
   448→  }
   449→
   450→  // Add reminder
   451→  const handleAddReminder = async () => {
   452→    if (!newReminder.trim() || !newReminderDue) return
   453→    try {
   454→      const res = await fetch('/api/crm/reminders', {
   455→        method: 'POST',
   456→        headers: { 'Content-Type': 'application/json' },
   457→        body: JSON.stringify({
   458→          leadId: lead.id,
   459→          userId: user?.id || 'demo',
   460→          title: newReminder.trim(),
   461→          dueDate: newReminderDue,
   462→        }),
   463→      })
   464→      if (res.ok) {
   465→        const data = await res.json()
   466→        setReminders(prev => [data.reminder, ...prev])
   467→        setNewReminder('')
   468→        setNewReminderDue('')
   469→        toast({ title: 'Reminder set!' })
   470→      }
   471→    } catch {
   472→      toast({ title: 'Failed to add reminder', variant: 'destructive' })
   473→    }
   474→  }
   475→
   476→  // Complete reminder
   477→  const handleCompleteReminder = async (reminderId: string) => {
   478→    try {
   479→      const res = await fetch('/api/crm/reminders', {
   480→        method: 'PATCH',
   481→        headers: { 'Content-Type': 'application/json' },
   482→        body: JSON.stringify({ reminderId, completed: true }),
   483→      })
   484→      if (res.ok) {
   485→        setReminders(prev => prev.map(r => r.id === reminderId ? { ...r, completed: true } : r))
   486→        toast({ title: 'Reminder completed' })
   487→      }
   488→    } catch {
   489→      toast({ title: 'Failed to update reminder', variant: 'destructive' })
   490→    }
   491→  }
   492→
   493→  // Delete reminder
   494→  const handleDeleteReminder = async (reminderId: string) => {
   495→    try {
   496→      const res = await fetch(`/api/crm/reminders?reminderId=${reminderId}`, { method: 'DELETE' })
   497→      if (res.ok) {
   498→        setReminders(prev => prev.filter(r => r.id !== reminderId))
   499→        toast({ title: 'Reminder deleted' })
   500→      }
   501→    } catch {
   502→      toast({ title: 'Failed to delete reminder', variant: 'destructive' })
   503→    }
   504→  }
   505→
   506→  const priority = PRIORITY_CONFIG[lead.priority] || PRIORITY_CONFIG.medium
   507→  const biz = lead.business
   508→
   509→  return (
   510→    <motion.div
   511→      initial={{ opacity: 0, x: 40 }}
   512→      animate={{ opacity: 1, x: 0 }}
   513→      exit={{ opacity: 0, x: 40 }}
   514→      className="fixed inset-y-0 right-0 z-50 w-full sm:w-[480px] bg-white border-l border-slate-200 shadow-2xl flex flex-col"
   515→    >
   516→      {/* Header */}
   517→      <div className="flex items-center justify-between p-4 border-b bg-slate-50">
   518→        <div className="flex-1 min-w-0">
   519→          <h2 className="text-lg font-bold text-slate-900 truncate">{biz.name}</h2>
   520→          <div className="flex items-center gap-2 mt-0.5 flex-wrap">
   521→            <Badge variant="outline" className="text-[10px] h-5 px-1.5">
   522→              <Tag className="h-2.5 w-2.5 mr-0.5" /> {biz.category}
   523→            </Badge>
   524→            {biz.city && (
   525→              <span className="text-xs text-muted-foreground flex items-center gap-0.5">
   526→                <MapPin className="h-3 w-3" /> {biz.city}
   527→              </span>
   528→            )}
   529→          </div>
   530→        </div>
   531→        <Button variant="ghost" size="icon" onClick={onClose} className="shrink-0">
   532→          <X className="h-5 w-5" />
   533→        </Button>
   534→      </div>
   535→
   536→      {/* Tabs */}
   537→      <Tabs value={activeTab} onValueChange={setActiveTab} className="flex-1 flex flex-col min-h-0">
   538→        <div className="border-b px-4">
   539→          <TabsList className="h-10 bg-transparent p-0 gap-4">
   540→            <TabsTrigger value="details" className="text-xs data-[state=active]:shadow-none border-b-2 border-transparent data-[state=active]:border-amber-500 data-[state=active]:bg-transparent rounded-none px-0 pb-0">Details</TabsTrigger>
   541→            <TabsTrigger value="notes" className="text-xs data-[state=active]:shadow-none border-b-2 border-transparent data-[state=active]:border-amber-500 data-[state=active]:bg-transparent rounded-none px-0 pb-0">
   542→              Notes{notes.length > 0 && ` (${notes.length})`}
   543→            </TabsTrigger>
   544→            <TabsTrigger value="tasks" className="text-xs data-[state=active]:shadow-none border-b-2 border-transparent data-[state=active]:border-amber-500 data-[state=active]:bg-transparent rounded-none px-0 pb-0">
   545→              Tasks{tasks.length > 0 && ` (${tasks.filter(t => !t.completed).length})`}
   546→            </TabsTrigger>
   547→            <TabsTrigger value="reminders" className="text-xs data-[state=active]:shadow-none border-b-2 border-transparent data-[state=active]:border-amber-500 data-[state=active]:bg-transparent rounded-none px-0 pb-0">
   548→              Reminders{reminders.length > 0 && ` (${reminders.filter(r => !r.completed).length})`}
   549→            </TabsTrigger>
   550→            <TabsTrigger value="activity" className="text-xs data-[state=active]:shadow-none border-b-2 border-transparent data-[state=active]:border-amber-500 data-[state=active]:bg-transparent rounded-none px-0 pb-0">Activity</TabsTrigger>
