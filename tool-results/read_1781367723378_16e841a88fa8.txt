   150→

   152→  const effectiveCategory = category === 'Other' ? customCategory : category
   153→

   155→    if (!effectiveCountry && !customCountry) {
   156→      toast({ title: 'Please select or enter a country', variant: 'destructive' })
   157→      return

   159→    if (!effectiveCategory && !customCategory) {
   160→      toast({ title: 'Please select or enter a business category', variant: 'destructive' })

   163→
   164→    setLoading(true)
   165→    setSearched(true)

   169→
   170→    // Progress timer
   171→    const progressInterval = setInterval(() => {
   172→      setElapsedTime(prev => prev + 1)
   173→    }, 1000)
   174→
   175→    // Progress messages rotation
   176→    const progressMessages = [
   177→      'Searching business directories...',
   178→      'Scanning Google Maps listings...',
   179→      'Checking Justdial & Sulekha...',
   180→      'Extracting business details...',
   181→      'AI analyzing results...',
   182→      'Scoring leads...',
   183→    ]
   184→    let msgIndex = 0
   185→    const msgInterval = setInterval(() => {
   186→      msgIndex = (msgIndex + 1) % progressMessages.length
   187→      setSearchProgress(progressMessages[msgIndex])
   188→    }, 5000)
   189→
   190→    try {
   191→      // Use AbortController with a generous 2-minute timeout
   192→      const controller = new AbortController()
   193→      const timeoutId = setTimeout(() => controller.abort(), 120000)
   194→
   195→      const res = await fetch('/api/businesses/search', {
   196→        method: 'POST',
   197→        headers: { 'Content-Type': 'application/json' },
   198→        body: JSON.stringify({
   199→          country: effectiveCountry,
   200→          state: state.trim(),
   201→          city: city.trim(),
   202→          category: effectiveCategory,
   203→          userId: user?.id || '',
   204→        }),
   205→        signal: controller.signal,
   206→      })
   207→
   208→      clearTimeout(timeoutId)
   209→
   210→      const data = res.ok ? await res.json() : null
   211→      const businesses = data?.businesses || []
   212→      const jobInfo = data?.searchJob || null
   213→
   214→      setResults(businesses)
   215→      setSearchJobInfo(jobInfo)
   216→

   218→        if (jobInfo?.fallback) {
   219→          toast({
   220→            title: 'Showing Cached Results',
   221→            description: `Found ${businesses.length} businesses from database. Live search is rate-limited — try again in 30s for fresh results.`,
   222→            duration: 6000,
   223→          })
   224→        } else {
   225→          toast({
   226→            title: 'Search Complete!',
   227→            description: `Found ${businesses.length} businesses in ${elapsedTime + 1}s. ${jobInfo?.duplicatesFound || 0} duplicates were merged.`,
   228→          })
   229→        }
   230→      } else if (jobInfo?.error) {
   231→        toast({
   232→          title: 'Search Temporarily Unavailable',
   233→          description: jobInfo.error || 'Please try again in a few seconds.',
   234→          variant: 'destructive',
   235→          duration: 5000,
   236→        })
   237→      } else {
   238→        toast({
   239→          title: 'No Results Found',
   240→          description: 'Try a different category or location.',
   241→          duration: 4000,
   242→        })
   243→      }

   246→        toast({
   247→          title: 'Search Timed Out',
   248→          description: 'The search is taking too long. Please try again.',
   249→          variant: 'destructive',
   250→          duration: 5000,
   251→        })
   252→      } else {
   253→        toast({
   254→          title: 'Search Failed',
   255→          description: 'Network error. Please try again.',
   256→          variant: 'destructive',
   257→        })

   314→        })

   317→      if (business.id) {

   411→
   412→  const filteredResults = showNoWebsiteOnly
   413→    ? results.filter((b) => b.websiteStatus !== 'HAS_WEBSITE')

   415→
   416→  const noWebsiteCount = results.filter((b) => !b.hasWebsite).length

   418→
   419→  const openDetail = (business: Business) => {

   422→
   423→  return (
   424→    <div className="space-y-6">

   429→
   430→      {/* Search Form */}

   562→      {/* Loading State */}

   599→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
