   130→   130→
   131→   131→function formatDate(dateStr: string): string {
   132→   132→  try {
   133→   133→    return new Date(dateStr).toLocaleDateString('en-US', {
   134→   134→      year: 'numeric',
   135→   135→      month: 'short',
   136→   136→      day: 'numeric',
   137→   137→    })
   138→   138→  } catch {
   139→   139→    return dateStr
   140→   140→  }
   141→   141→}
   142→   142→
   143→   143→function formatCurrency(amount: number): string {
   144→   144→  return new Intl.NumberFormat('en-US', {
   145→   145→    style: 'currency',
   146→   146→    currency: 'USD',
   147→   147→  }).format(amount)
   148→   148→}
   149→   149→
   150→   150→function formatExpiry(month: number, year: number): string {
   151→   151→  return `${String(month).padStart(2, '0')}/${year}`
   152→   152→}
