   137→function parseFeatures(featuresJson: string): PlanFeature[] {
   138→  try {
   139→    const parsed = JSON.parse(featuresJson)
   140→    if (Array.isArray(parsed)) {
   141→      return parsed.map((f: string | PlanFeature) => {
   142→        if (typeof f === 'string') {
   143→          return { label: f, included: true }
   144→        }
   145→        return f as PlanFeature
   146→      })
   147→    }
   148→    return []
   149→  } catch {
   150→    return []
   151→  }
   152→}
   153→
   154→function getPlanIcon(name: string): React.ReactNode {
   155→  const lower = name.toLowerCase()
   156→  if (lower.includes('enterprise') || lower.includes('team') || lower.includes('agency')) {
   157→    return <Globe className="h-5 w-5" />
   158→  }
   159→  if (lower.includes('pro') || lower.includes('professional')) {
   160→    return <Crown className="h-5 w-5" />
   161→  }
   162→  return <Zap className="h-5 w-5" />
   163→}
   164→
   165→function formatNumber(n: number): string {
   166→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k`
   167→  return n.toString()
   168→}
   169→
   170→function getUsagePercentage(used: number, total: number): number {
   171→  if (total === 0) return 0
   172→  return Math.min(Math.round((used / total) * 100), 100)
   173→}
   174→
   175→function getUsageColor(percent: number): string {
   176→  if (percent >= 90) return 'bg-red-500'
   177→  if (percent >= 70) return 'bg-amber-500'
   178→  return 'bg-emerald-500'
   179→}
   180→
   181→function formatDate(dateStr: string): string {
   182→  try {
   183→    const date = new Date(dateStr)
   184→    return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
   185→  } catch {
   186→    return dateStr
   187→  }
   188→}
   189→
   190→function formatCurrency(amount: number): string {
   191→  return `$${Math.abs(amount).toFixed(2)}`
   192→}
   193→
   194→function getBillingStatusBadge(type: string, amount: number) {
   195→  if (type === 'refund') {
   196→    return <Badge className="bg-emerald-50 text-emerald-700 hover:bg-emerald-50 border-0 text-xs">Refund</Badge>
   197→  }
   198→  if (type === 'purchase') {
   199→    return <Badge className="bg-blue-50 text-blue-700 hover:bg-blue-50 border-0 text-xs">Purchase</Badge>
   200→  }
   201→  if (type === 'subscription') {
   202→    return <Badge className="bg-amber-50 text-amber-700 hover:bg-amber-50 border-0 text-xs">Subscription</Badge>
   203→  }
   204→  return <Badge className="bg-slate-50 text-slate-700 hover:bg-slate-50 border-0 text-xs">{type}</Badge>
   205→}

   301→export function SubscriptionView() {
   302→  const { user } = useAppStore()
   303→  const { toast } = useToast()
   304→
   305→  const [data, setData] = useState<SubscriptionApiData | null>(null)
   306→  const [loading, setLoading] = useState(true)
   307→  const [error, setError] = useState<string | null>(null)
   308→
   309→  const fetchSubscription = useCallback(async () => {
   310→    if (!user?.id) return
   311→    setLoading(true)
   312→    setError(null)
   313→    try {
   314→      const res = await fetch(`/api/user/subscription?userId=${user.id}`)
   315→      if (!res.ok) {
   316→        const body = await res.json().catch(() => ({}))
   317→        throw new Error(body.error || 'Failed to fetch subscription data')
   318→      }
   319→      const json = await res.json()
   320→      setData(json)
   321→    } catch (err: any) {
   322→      console.error('Failed to fetch subscription:', err)
   323→      setError(err.message || 'Failed to load subscription data')
   324→      toast({
   325→        title: 'Error',
   326→        description: 'Failed to load subscription data. Please try again.',
   327→        variant: 'destructive',
   328→      })
   329→    } finally {
   330→      setLoading(false)
   331→    }
   332→  }, [user?.id, toast])
   333→
   334→  useEffect(() => {
   335→    fetchSubscription()
   336→  }, [fetchSubscription])
   337→
   338→  if (loading) return <SubscriptionSkeleton />
   339→
   340→  if (error && !data) {
   341→    return (
   342→      <div className="space-y-8 max-w-6xl">
   343→        <div>
   344→          <h1 className="text-2xl font-bold text-slate-900">Subscription</h1>
   345→          <p className="text-muted-foreground">Manage your plan, usage, and billing</p>
   346→        </div>
   347→        <Card className="border-0 shadow-sm">
   348→          <CardContent className="p-12 flex flex-col items-center text-center">
   349→            <div className="h-14 w-14 rounded-2xl bg-red-50 flex items-center justify-center mb-4">
   350→              <AlertCircle className="h-7 w-7 text-red-500" />
   351→            </div>
   352→            <h3 className="text-lg font-semibold text-slate-700 mb-1">Failed to load subscription data</h3>
   353→            <p className="text-sm text-muted-foreground mb-4">{error}</p>
   354→            <Button onClick={fetchSubscription} variant="outline">
   355→              Try Again
   356→            </Button>
   357→          </CardContent>
   358→        </Card>
   359→      </div>
   360→    )
   361→  }
   362→
   363→  if (!data) return null
   364→
   365→  // Build display plans from API data
   366→  const currentPlanId = data.currentPlan?.id ?? user?.planId ?? ''

   369→    const nameLower = plan.name.toLowerCase()

   371→
   372→    return {
   373→      id: plan.id,
   374→      name: plan.name,
   375→      price: plan.price,
   376→      period: '/mo',
   377→      description: plan.description,
   378→      icon: getPlanIcon(plan.name),
   379→      features: parseFeatures(plan.features),
   380→      highlight: isCurrent || plan.popular,
   381→      current: isCurrent,
   382→      enterprise: isEnterprise,
   383→    }
   384→  })
   385→

   387→  const usageItems = [
   388→    {
   389→      label: 'Leads',
   390→      used: data.usage.leads.used,
   391→      total: data.usage.leads.limit,

   394→    },

   398→      total: data.usage.searches.limit,
   399→      icon: <BarChart3 className="h-4 w-4" />,
   400→      color: 'text-emerald-600',
   401→    },
   402→    {
   403→      label: 'Exports',
   404→      used: data.usage.exports.used,
   405→      total: data.usage.exports.limit,
   406→      icon: <Code className="h-4 w-4" />,
   407→      color: 'text-violet-600',
   408→    },
   409→    {
   410→      label: 'Credits',
   411→      used: data.usage.credits.total - data.usage.credits.remaining,
   412→      total: data.usage.credits.total,
   413→      icon: <Zap className="h-4 w-4" />,
   414→      color: 'text-orange-600',
   415→    },
   416→  ]
   417→
   418→  const currentPlan = data.currentPlan
   419→  const subscription = data.subscription
