   103→// ─── Plan Icon Mapping ──────────────────────────────────────────────────────
   104→
   105→function getPlanIcon(tier: string, className?: string) {
   106→  switch (tier) {
   107→    case 'starter':
   108→      return <Rocket className={className || 'h-6 w-6'} />
   109→    case 'agency':
   110→      return <Zap className={className || 'h-6 w-6'} />
   111→    case 'enterprise':
   112→      return <Crown className={className || 'h-6 w-6'} />
   113→    default:
   114→      return <Star className={className || 'h-6 w-6'} />
   115→  }
   116→}
   117→
   118→function getPlanIconBg(tier: string) {
   119→  switch (tier) {
   120→    case 'starter':
   121→      return 'bg-slate-100 text-slate-600'
   122→    case 'agency':
   123→      return 'bg-amber-100 text-amber-600'
   124→    case 'enterprise':
   125→      return 'bg-emerald-100 text-emerald-600'
   126→    default:
   127→      return 'bg-slate-100 text-slate-600'
   128→  }
   129→}
   130→
   131→function getPlanCardBorder(tier: string, popular: boolean) {
   132→  if (popular) return 'ring-2 ring-amber-400 shadow-lg shadow-amber-500/10'
   133→  switch (tier) {
   134→    case 'starter':
   135→      return 'border-slate-200'
   136→    case 'agency':
   137→      return 'border-amber-200'
   138→    case 'enterprise':
   139→      return 'border-emerald-200'
   140→    default:
   141→      return ''
   142→  }
   143→}
   144→
   145→function getPlanButtonStyle(tier: string, popular: boolean, isCurrent: boolean) {
   146→  if (isCurrent) return 'bg-slate-100 text-slate-500 cursor-default'
   147→  if (popular) return 'bg-gradient-to-r from-amber-500 to-orange-600 text-white hover:from-amber-600 hover:to-orange-700 shadow-lg shadow-amber-500/25'
   148→  switch (tier) {
   149→    case 'starter':
   150→      return 'bg-slate-800 text-white hover:bg-slate-900'
   151→    case 'agency':
   152→      return 'bg-amber-500 text-white hover:bg-amber-600'
   153→    case 'enterprise':
   154→      return 'bg-emerald-600 text-white hover:bg-emerald-700'
   155→    default:
   156→      return ''
   157→  }
   158→}
   159→
   160→function formatCredits(credits: number): string {
   161→  if (credits === -1 || credits >= 999999) return 'Unlimited'
   162→  return credits.toLocaleString('en-IN')
   163→}

   196→            id: data.subscription.id,

   200→            cancelAtPeriodEnd: data.subscription.cancelAtPeriodEnd || false,

   202→              id: data.currentPlan?.id || '',

   204→              tier: data.currentPlan?.tier || 'free',
   205→              price: data.currentPlan?.price || 0,
   206→            },
   207→          })
   208→        }
   209→      }
   210→    } catch {
   211→      // Silent fail
   212→    } finally {
   213→      setLoading(false)
   214→    }

   227→        method: 'POST',
   228→        headers: { 'Content-Type': 'application/json' },

   230→          userId: user.id,
   231→          tier,
   232→          interval: billingInterval,
   233→        }),

   235→
   236→      const data = await res.json()
   237→
   238→      if (res.ok) {
   239→        if (data.demo) {
   240→          toast({
   241→            title: 'Subscription Activated!',
   242→            description: `${tier.charAt(0).toUpperCase() + tier.slice(1)} plan (${billingInterval}) is now active. Credits have been added to your account.`,

   244→          // Refresh data
   245→          fetchData()
   246→        } else if (data.url) {
   247→          // Redirect to Stripe Checkout
   248→          window.location.href = data.url
   249→        }
   250→      } else {
   251→        toast({
   252→          title: 'Error',
   253→          description: data.error || 'Failed to start checkout',
   254→          variant: 'destructive',
   255→        })
   256→      }
   257→    } catch {
   258→      toast({
   259→        title: 'Error',
   260→        description: 'Something went wrong. Please try again.',
   261→        variant: 'destructive',
   262→      })
   263→    } finally {
   264→      setSubscribing(null)

   272→    try {

   274→        method: 'POST',
   275→        headers: { 'Content-Type': 'application/json' },
   276→        body: JSON.stringify({ userId: user.id, immediately }),
   277→      })
   278→
   279→      if (res.ok) {
   280→        toast({
   281→          title: immediately ? 'Subscription Canceled' : 'Scheduled Cancellation',
   282→          description: immediately
   283→            ? 'Your subscription has been canceled.'
   284→            : 'Your subscription will be canceled at the end of the billing period.',
   285→        })
   286→        fetchData()
   287→      }
   288→    } catch {
   289→      toast({ title: 'Error', description: 'Failed to cancel subscription', variant: 'destructive' })

   291→      setCanceling(false)
   292→      setCancelDialog(false)
   293→    }

   334→          <div className="h-10 w-10 rounded-xl bg-amber-100 flex items-center justify-center">
   335→            <Crown className="h-5 w-5 text-amber-600" />

   356→                    <div className="flex items-center gap-2">
   357→                      <h3 className="font-semibold text-lg">
   358→                        {subscription.plan.name} Plan
   359→                      </h3>
   360→                      <Badge variant="outline" className={subscription.cancelAtPeriodEnd

   385→                  <Button

   387→                    size="sm"
   388→                    onClick={async () => {
   389→                      const res = await fetch('/api/stripe/portal', {
   390→                        method: 'POST',
   391→                        headers: { 'Content-Type': 'application/json' },
   392→                        body: JSON.stringify({ userId: user?.id }),

   394→                      const data = await res.json()

   396→                        window.location.href = data.url
   397→                      } else {
   398→                        toast({
   399→                          title: 'Portal Not Available',
   400→                          description: data.message || 'Manage your subscription from the billing page.',
   401→                        })
   402→                      }
   403→                    }}
   404→                  >
   405→                    Manage
   406→                  </Button>
   407→                </div>
   408→              </div>
   409→            </CardContent>
   410→          </Card>
   411→        </motion.div>

   414→      {/* Billing interval toggle */}
