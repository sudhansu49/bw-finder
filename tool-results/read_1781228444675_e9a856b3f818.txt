   149→function formatCurrency(value: number): string {
   150→  if (value >= 1000000) return `$${(value / 1000000).toFixed(1)}M`
   151→  if (value >= 1000) return `$${(value / 1000).toFixed(0)}k`
   152→  return `$${value.toLocaleString()}`
   153→}
   154→
   155→function getScoreColor(score: number | null | undefined): { bg: string; text: string } {
   156→  if (score === null || score === undefined) return { bg: 'bg-slate-100', text: 'text-slate-400' }
   157→  if (score >= 70) return { bg: 'bg-emerald-50', text: 'text-emerald-600' }
   158→  if (score >= 40) return { bg: 'bg-amber-50', text: 'text-amber-600' }
   159→  return { bg: 'bg-red-50', text: 'text-red-500' }
   160→}
