   132→export function SearchView() {
   133→  const { user, setCurrentView } = useAppStore()
   134→  const { toast } = useToast()
   135→  const [country, setCountry] = useState('')
   136→  const [state, setState] = useState('')
   137→  const [city, setCity] = useState('')
   138→  const [category, setCategory] = useState('')
   139→  const [customCountry, setCustomCountry] = useState('')
   140→  const [customCategory, setCustomCategory] = useState('')
   141→  const [loading, setLoading] = useState(false)
   142→  const [results, setResults] = useState<Business[]>([])
   143→  const [showNoWebsiteOnly, setShowNoWebsiteOnly] = useState(false)
   144→  const [searched, setSearched] = useState(false)
   145→  const [searchJobInfo, setSearchJobInfo] = useState<SearchJobInfo | null>(null)
   146→  const [selectedBusiness, setSelectedBusiness] = useState<Business | null>(null)
   147→  const [detailOpen, setDetailOpen] = useState(false)
   148→  const [addedLeads, setAddedLeads] = useState<Set<string>>(new Set())
   149→  const [addingLead, setAddingLead] = useState<string | null>(null)
   150→
   151→  const effectiveCountry = country === 'Other' ? customCountry : country
   152→  const effectiveCategory = category === 'Other' ? customCategory : category
   153→
   154→  const handleSearch = async () => {
   155→    if (!effectiveCountry && !customCountry) {
   156→      toast({ title: 'Please select or enter a country', variant: 'destructive' })
   157→      return
   158→    }
   159→    if (!effectiveCategory && !customCategory) {
   160→      toast({ title: 'Please select or enter a business category', variant: 'destructive' })
   161→      return
   162→    }
   163→
   164→    setLoading(true)
   165→    setSearched(true)
   166→    setSearchJobInfo(null)
   167→
   168→    try {
   169→      const res = await fetch('/api/businesses/search', {
   170→        method: 'POST',
   171→        headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({
   173→          country: effectiveCountry,
   174→          state: state.trim(),
   175→          city: city.trim(),
   176→          category: effectiveCategory,
   177→          userId: user?.id || 'demo',
   178→        }),
   179→      })
   180→
   181→      const data = res.ok ? await res.json() : null
   182→      const businesses = data?.businesses || []
   183→      const jobInfo = data?.searchJob || null
   184→
   185→      setResults(businesses)
   186→      setSearchJobInfo(jobInfo)
   187→
   188→      if (businesses.length > 0) {
   189→        if (jobInfo?.fallback) {
   190→          toast({
   191→            title: 'Showing Cached Results',
   192→            description: `Found ${businesses.length} businesses from database. Live search is rate-limited — try again in 30s for fresh results.`,
   193→            duration: 6000,
   194→          })
   195→        } else {
   196→          toast({
   197→            title: 'Search Complete!',
   198→            description: `Found ${businesses.length} businesses. ${jobInfo?.duplicatesFound || 0} duplicates were merged.`,
   199→          })
   200→        }
   201→      } else if (jobInfo?.error) {
   202→        toast({
   203→          title: 'Search Temporarily Unavailable',
   204→          description: jobInfo.error || 'Please try again in a few seconds.',
   205→          variant: 'destructive',
   206→          duration: 5000,
   207→        })
   208→      } else {
   209→        toast({
   210→          title: 'No Results Found',
   211→          description: 'Try a different category or location.',
   212→          duration: 4000,
   213→        })
   214→      }
   215→    } catch {
   216→      toast({
   217→        title: 'Search Failed',
   218→        description: 'Network error. Please try again.',
   219→        variant: 'destructive',
   220→      })
   221→      setResults([])
   222→    } finally {
   223→      setLoading(false)
   224→    }
   225→  }
   226→
   227→  const handleAddAsLead = async (business: Business) => {
   228→    if (!business.id || addedLeads.has(business.id)) return
   229→
   230→    setAddingLead(business.id || business.name)
   231→    try {
   232→      let businessId = business.id
   233→
   234→      if (!businessId) {
   235→        const businessRes = await fetch('/api/businesses', {
   236→          method: 'POST',
   237→          headers: { 'Content-Type': 'application/json' },
   238→          body: JSON.stringify({
   239→            name: business.name,
   240→            category: business.category,
   241→            address: business.address || '',
   242→            city: business.city || '',
   243→            state: business.state || '',
   244→            country: business.country || '',
   245→            phone: business.phone || '',
   246→            email: business.email || '',
   247→            website: business.website || null,
   248→            hasWebsite: business.hasWebsite,
   249→            facebookUrl: business.facebookUrl || null,
   250→            instagramUrl: business.instagramUrl || null,
   251→            linkedinUrl: business.linkedinUrl || null,
   252→          }),
   253→        })
   254→        if (businessRes.ok) {
   255→          const businessData = await businessRes.json()
   256→          businessId = businessData.business?.id || businessData.id
   257→        }
   258→      }
   259→
   260→      if (businessId) {
   261→        await fetch('/api/leads', {
   262→          method: 'POST',
   263→          headers: { 'Content-Type': 'application/json' },
   264→          body: JSON.stringify({
   265→            businessId,
   266→            userId: user?.id || 'demo',
   267→            status: 'New',
   268→            priority: business.hasWebsite ? 'low' : 'medium',
   269→            estimatedValue: business.hasWebsite ? 800 : 1500,
   270→            notes: `Lead from discovery search: ${business.name} in ${business.city || ''}, ${business.country || ''}`,
   271→          }),
   272→        })
   273→      }
   274→
   275→      if (business.id) {
   276→        setAddedLeads(prev => new Set([...prev, business.id!]))
   277→      }
   278→
   279→      toast({
   280→        title: 'Lead Added!',
   281→        description: `${business.name} has been added to your leads pipeline.`,
   282→      })
   283→    } catch {
   284→      toast({
   285→        title: 'Error',
   286→        description: `Failed to add ${business.name} as lead.`,
   287→        variant: 'destructive',
   288→      })
   289→    } finally {
   290→      setAddingLead(null)
   291→    }
   292→  }
   293→
   294→  const handleAddAllNoWebsiteAsLeads = async () => {
   295→    const noWebsiteBusinesses = filteredResults.filter(b => !b.hasWebsite && b.id && !addedLeads.has(b.id))
   296→    let added = 0
   297→
   298→    for (const business of noWebsiteBusinesses) {
   299→      try {
   300→        await fetch('/api/leads', {
   301→          method: 'POST',
   302→          headers: { 'Content-Type': 'application/json' },
   303→          body: JSON.stringify({
   304→            businessId: business.id,
   305→            userId: user?.id || 'demo',
   306→            status: 'New',
   307→            priority: 'medium',
   308→            estimatedValue: 1500,
   309→            notes: `Bulk lead from discovery: ${business.name}`,
   310→          }),
   311→        })
   312→        if (business.id) {
   313→          setAddedLeads(prev => new Set([...prev, business.id!]))
   314→        }
   315→        added++
   316→      } catch {
   317→        // Skip failed ones
   318→      }
   319→    }
   320→
   321→    toast({
   322→      title: `${added} Leads Added!`,
   323→      description: `Added ${added} businesses without websites to your leads pipeline.`,
   324→    })
   325→  }
   326→
   327→  // Website status badge component
   328→  const WebsiteStatusBadge = ({ status, hasWebsite }: { status?: string | null, hasWebsite: boolean }) => {
   329→    if (status === 'HAS_WEBSITE') {
   330→      return (
   331→        <Badge className="bg-emerald-100 text-emerald-800 hover:bg-emerald-100 text-xs font-semibold border border-emerald-200">
   332→          <Globe className="h-3 w-3 mr-1" />
   333→          Website Exists
   334→        </Badge>
   335→      )
   336→    }
   337→    if (status === 'SOCIAL_ONLY') {
   338→      return (
   339→        <Badge className="bg-red-100 text-red-800 hover:bg-red-100 text-xs font-semibold border border-red-200">
   340→          <Unplug className="h-3 w-3 mr-1" />
   341→          Social Only
   342→        </Badge>
   343→      )
   344→    }
   345→    // NO_WEBSITE or not detected yet
   346→    return (
   347→      <Badge className="bg-red-100 text-red-800 hover:bg-red-100 text-xs font-semibold border border-red-200">
   348→        <Unplug className="h-3 w-3 mr-1" />
   349→        No Website
   350→      </Badge>
   351→    )
   352→  }
   353→
   354→  // Lead score color
   355→  const getScoreColor = (score: number | null | undefined) => {
   356→    if (!score) return 'text-slate-400'
   357→    if (score >= 70) return 'text-emerald-600'
   358→    if (score >= 40) return 'text-amber-600'
   359→    return 'text-red-500'
   360→  }
   361→
   362→  const getScoreBg = (score: number | null | undefined) => {
   363→    if (!score) return 'bg-slate-50'
   364→    if (score >= 70) return 'bg-emerald-50'
   365→    if (score >= 40) return 'bg-amber-50'
   366→    return 'bg-red-50'
   367→  }
   368→
   369→  const filteredResults = showNoWebsiteOnly
   370→    ? results.filter((b) => b.websiteStatus !== 'HAS_WEBSITE')
   371→    : results
   372→
   373→  const noWebsiteCount = results.filter((b) => !b.hasWebsite).length
   374→  const withWebsiteCount = results.filter((b) => b.hasWebsite).length
   375→
   376→  const openDetail = (business: Business) => {
   377→    setSelectedBusiness(business)
   378→    setDetailOpen(true)
   379→  }
   380→
   381→  return (
   382→    <div className="space-y-6">
   383→      <div>
   384→        <h1 className="text-2xl font-bold text-slate-900">Lead Discovery Engine</h1>
   385→        <p className="text-muted-foreground">Find local businesses without websites across any location</p>
   386→      </div>
   387→
   388→      {/* Search Form */}
   389→      <Card className="border-0 shadow-sm">
   390→        <CardHeader className="pb-3">
   391→          <CardTitle className="text-base flex items-center gap-2">
   392→            <Search className="h-4 w-4 text-amber-500" />
   393→            Search Criteria
   394→          </CardTitle>
   395→        </CardHeader>
   396→        <CardContent className="p-6 pt-0">
   397→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   398→            {/* Country */}
   399→            <div className="space-y-2">
   400→              <Label htmlFor="country" className="text-sm font-medium">
   401→                Country <span className="text-red-500">*</span>
   402→              </Label>
   403→              <Select value={country} onValueChange={setCountry}>
   404→                <SelectTrigger className="h-11">
   405→                  <SelectValue placeholder="Select country" />
   406→                </SelectTrigger>
   407→                <SelectContent>
   408→                  {countries.map((c) => (
   409→                    <SelectItem key={c} value={c}>
   410→                      {c}
   411→                    </SelectItem>
   412→                  ))}
   413→                </SelectContent>
   414→              </Select>
   415→              {country === 'Other' && (
   416→                <Input
   417→                  placeholder="Enter country name"
   418→                  value={customCountry}
   419→                  onChange={(e) => setCustomCountry(e.target.value)}
   420→                  className="h-10"
   421→                />
   422→              )}
   423→            </div>
   424→
   425→            {/* State */}
   426→            <div className="space-y-2">
   427→              <Label htmlFor="state" className="text-sm font-medium">State / Province</Label>
   428→              <div className="relative">
   429→                <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   430→                <Input
   431→                  id="state"
   432→                  placeholder="e.g. Maharashtra, California"
   433→                  value={state}
   434→                  onChange={(e) => setState(e.target.value)}
   435→                  className="pl-10 h-11"
   436→                />
   437→              </div>
   438→            </div>
   439→
   440→            {/* City */}
   441→            <div className="space-y-2">
   442→              <Label htmlFor="city" className="text-sm font-medium">City</Label>
   443→              <div className="relative">
   444→                <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   445→                <Input
   446→                  id="city"
   447→                  placeholder="e.g. Mumbai, New York"
   448→                  value={city}
   449→                  onChange={(e) => setCity(e.target.value)}
   450→                  className="pl-10 h-11"
   451→                />
   452→              </div>
   453→            </div>
   454→
   455→            {/* Category */}
   456→            <div className="space-y-2">
   457→              <Label className="text-sm font-medium">
   458→                Business Category <span className="text-red-500">*</span>
   459→              </Label>
   460→              <Select value={category} onValueChange={setCategory}>
   461→                <SelectTrigger className="h-11">
   462→                  <SelectValue placeholder="Select category" />
   463→                </SelectTrigger>
   464→                <SelectContent>
   465→                  {categories.map((cat) => (
   466→                    <SelectItem key={cat} value={cat}>
   467→                      {cat}
   468→                    </SelectItem>
   469→                  ))}
   470→                </SelectContent>
   471→              </Select>
   472→              {category === 'Other' && (
   473→                <Input
   474→                  placeholder="Enter business category"
   475→                  value={customCategory}
   476→                  onChange={(e) => setCustomCategory(e.target.value)}
   477→                  className="h-10"
   478→                />
   479→              )}
   480→            </div>
   481→          </div>
   482→
   483→          {/* Search Button */}
   484→          <div className="mt-5 flex items-center gap-3">
   485→            <Button
   486→              onClick={handleSearch}
   487→              disabled={loading || !effectiveCountry || !effectiveCategory}
   488→              className="h-11 px-8 bg-amber-500 hover:bg-amber-600 text-white font-semibold"
   489→            >
   490→              {loading ? (
   491→                <>
   492→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   493→                  Discovering Businesses...
   494→                </>
   495→              ) : (
   496→                <>
   497→                  <Search className="mr-2 h-4 w-4" />
   498→                  Discover Businesses
   499→                </>
   500→              )}
   501→            </Button>
   502→            {searched && !loading && (
   503→              <Button
   504→                variant="outline"
   505→                onClick={() => {
   506→                  setSearched(false)
   507→                  setResults([])
   508→                  setSearchJobInfo(null)
   509→                  setAddedLeads(new Set())
   510→                }}
   511→                className="h-11"
   512→              >
   513→                Clear Results
   514→              </Button>
   515→            )}
   516→          </div>
   517→        </CardContent>
   518→      </Card>
   519→
   520→      {/* Loading State */}
   521→      {loading && (
   522→        <motion.div
   523→          initial={{ opacity: 0 }}
   524→          animate={{ opacity: 1 }}
   525→          className="flex flex-col items-center justify-center py-16"
   526→        >
   527→          <div className="relative">
   528→            <div className="animate-spin rounded-full h-16 w-16 border-4 border-amber-200 border-t-amber-500" />
   529→            <Search className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 h-6 w-6 text-amber-500" />
   530→          </div>
   531→          <p className="mt-4 text-muted-foreground font-medium">Discovering businesses...</p>
   532→          <p className="text-sm text-muted-foreground mt-1">Searching multiple directories and business listings</p>
   533→          <div className="mt-4 flex items-center gap-2 text-xs text-muted-foreground">
   534→            <div className="flex items-center gap-1">
   535→              <div className="h-2 w-2 rounded-full bg-amber-500 animate-pulse" />
   536→              Web Search
   537→            </div>
   538→            <div className="flex items-center gap-1">
   539→              <div className="h-2 w-2 rounded-full bg-emerald-500 animate-pulse" />
   540→              Directory Scan
   541→            </div>
   542→            <div className="flex items-center gap-1">
   543→              <div className="h-2 w-2 rounded-full bg-blue-500 animate-pulse" />
   544→              Data Extraction
   545→            </div>
   546→          </div>
   547→        </motion.div>
   548→      )}
   549→
   550→      {/* Results */}
   551→      {!loading && searched && (

   553→          {/* Results Header & Stats */}
   554→          <div className="flex flex-col gap-4">
   555→            {/* Stats Row */}
   556→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   557→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   558→                <p className="text-2xl font-bold text-slate-900">{results.length}</p>
   559→                <p className="text-xs text-muted-foreground">Total Found</p>
   560→              </div>
   561→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   562→                <p className="text-2xl font-bold text-red-600">{noWebsiteCount}</p>
   563→                <p className="text-xs text-muted-foreground">No Website</p>
   564→              </div>
   565→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   566→                <p className="text-2xl font-bold text-emerald-600">{withWebsiteCount}</p>
   567→                <p className="text-xs text-muted-foreground">Has Website</p>
   568→              </div>
   569→              <div className="bg-white rounded-xl border-0 shadow-sm p-4 text-center">
   570→                <p className="text-2xl font-bold text-amber-600">
   571→                  {searchJobInfo?.duplicatesFound || 0}
   572→                </p>
   573→                <p className="text-xs text-muted-foreground">Duplicates Merged</p>
   574→              </div>
   575→            </div>
   576→
   577→            {/* Fallback / Search Info Banner */}
   578→            {searchJobInfo?.fallback && (
   579→              <div className="flex items-center gap-2 text-xs text-amber-700 bg-amber-50 border border-amber-200 rounded-lg px-4 py-2">
   580→                <AlertCircle className="h-3.5 w-3.5 shrink-0" />
   581→                <span>Live search rate-limited — showing <span className="font-semibold">cached results</span> from your database. Try again in 30 seconds for fresh results.</span>
   582→              </div>
   583→            )}
   584→            {searchJobInfo && searchJobInfo.sourcesUsed && !searchJobInfo.fallback && (
   585→              <div className="flex items-center gap-2 text-xs text-muted-foreground bg-amber-50 rounded-lg px-4 py-2">
   586→                <AlertCircle className="h-3.5 w-3.5 text-amber-600" />
   587→                <span>Sources searched: <span className="font-medium text-amber-700">{searchJobInfo.sourcesUsed}</span></span>
   588→              </div>
   589→            )}
   590→
   591→            {/* Actions Row */}
   592→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   593→              <div className="flex items-center gap-3">
   594→                <h2 className="text-lg font-semibold">
   595→                  {filteredResults.length} result{filteredResults.length !== 1 ? 's' : ''}
   596→                </h2>
   597→                <div className="flex items-center gap-2">
   598→                  <Button
   599→                    variant={showNoWebsiteOnly ? 'default' : 'outline'}
   600→                    size="sm"
   601→                    onClick={() => setShowNoWebsiteOnly(!showNoWebsiteOnly)}
