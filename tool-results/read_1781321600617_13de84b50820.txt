   237→   237→export function ProposalView() {
   238→   238→  const { user } = useAppStore()
   239→   239→  const { toast } = useToast()
   240→   240→  const [businesses, setBusinesses] = useState<Business[]>([])
   241→   241→  const [loading, setLoading] = useState(true)
   242→   242→  const [generating, setGenerating] = useState<string | null>(null)
   243→   243→  const [selectedBusiness, setSelectedBusiness] = useState<Business | null>(null)
   244→   244→  const [proposal, setProposal] = useState<ProposalData | null>(null)
   245→   245→  const [proposalOpen, setProposalOpen] = useState(false)
   246→   246→  const [loadingProposal, setLoadingProposal] = useState(false)
   247→   247→  const [searchTerm, setSearchTerm] = useState('')
   248→   248→  const [categoryFilter, setCategoryFilter] = useState('All')
   249→   249→  const [page, setPage] = useState(1)
   250→   250→  const pageSize = 8
   251→   251→
   252→   252→  const categories = ['All', 'Restaurant', 'Salon', 'Beauty Parlour', 'Spa', 'Gym', 'Clinic', 'Hotel', 'Real Estate', 'Dentist', 'Lawyer', 'School', 'Other']
   253→   253→
   254→   254→  // Fetch businesses
   255→   255→  useEffect(() => {
   256→   256→    const fetchBusinesses = async () => {
   257→   257→      try {
   258→   258→        const res = await fetch('/api/businesses?limit=200')
   259→   259→        if (res.ok) {
   260→   260→          const data = await res.json()
   261→   261→          setBusinesses(data.businesses || [])
   262→   262→        }
   263→   263→      } catch { setBusinesses([]) }
   264→   264→      finally { setLoading(false) }
   265→   265→    }
   266→   266→    fetchBusinesses()
   267→   267→  }, [user?.id])
   268→   268→
   269→   269→  const filteredBusinesses = businesses.filter(b => {
   270→   270→    const matchesSearch = !searchTerm || b.name.toLowerCase().includes(searchTerm.toLowerCase()) || b.city?.toLowerCase().includes(searchTerm.toLowerCase())
   271→   271→    const matchesCategory = categoryFilter === 'All' || b.category === categoryFilter
   272→   272→    return matchesSearch && matchesCategory
   273→   273→  })
   274→   274→
   275→   275→  const totalPages = Math.ceil(filteredBusinesses.length / pageSize)
   276→   276→  const paginatedBusinesses = filteredBusinesses.slice((page - 1) * pageSize, page * pageSize)
   277→   277→
   278→   278→  const proposalsGenerated = businesses.filter(b => b.proposalData).length
   279→   279→
   280→   280→  // Generate proposal
   281→   281→  const handleGenerateProposal = async (business: Business, useAI = false) => {
   282→   282→    setGenerating(business.id)
   283→   283→    try {
   284→   284→      const res = await fetch('/api/businesses/proposal', {
   285→   285→        method: 'POST',
   286→   286→        headers: { 'Content-Type': 'application/json' },
   287→   287→        body: JSON.stringify({ businessId: business.id, useAI }),
   288→   288→      })
   289→   289→      if (res.ok) {
   290→   290→        const data = await res.json()
   291→   291→        setProposal(data.proposal)
   292→   292→        setSelectedBusiness(business)
   293→   293→        setProposalOpen(true)
   294→   294→        toast({ title: 'Proposal Generated!', description: `Proposal for ${business.name} is ready.` })
   295→   295→        // Refresh businesses list
   296→   296→        const bizRes = await fetch('/api/businesses?limit=200')
   297→   297→        if (bizRes.ok) {
   298→   298→          const bizData = await bizRes.json()
   299→   299→          setBusinesses(bizData.businesses || [])
   300→   300→        }
   301→   301→      } else {
   302→   302→        toast({ title: 'Failed to generate proposal', variant: 'destructive' })
   303→   303→      }
   304→   304→    } catch {
   305→   305→      toast({ title: 'Error generating proposal', variant: 'destructive' })
   306→   306→    } finally {
   307→   307→      setGenerating(null)
   308→   308→    }
   309→   309→  }
   310→   310→
   311→   311→  // View existing proposal
   312→   312→  const handleViewProposal = async (business: Business) => {
   313→   313→    setSelectedBusiness(business)
   314→   314→    setLoadingProposal(true)
   315→   315→    setProposalOpen(true)
   316→   316→    try {
   317→   317→      const res = await fetch(`/api/businesses/proposal?businessId=${business.id}`)
   318→   318→      if (res.ok) {
   319→   319→        const data = await res.json()
   320→   320→        setProposal(data.proposal)
   321→   321→      } else {
   322→   322→        toast({ title: 'Failed to load proposal', variant: 'destructive' })
   323→   323→      }
   324→   324→    } catch {
   325→   325→      toast({ title: 'Error loading proposal', variant: 'destructive' })
   326→   326→    } finally {
   327→   327→      setLoadingProposal(false)
   328→   328→    }
   329→   329→  }
   330→   330→
   331→   331→  // Export PDF
   332→   332→  const handleExportPDF = async (businessId: string) => {
   333→   333→    try {
   334→   334→      toast({ title: 'Generating PDF...', description: 'Your proposal PDF is being prepared.' })

   336→   336→      if (res.ok) {
   337→   337→        const html = await res.text()
   338→   338→        // Open in new window for printing
   339→   339→        const printWindow = window.open('', '_blank')
   340→   340→        if (printWindow) {
   341→   341→          printWindow.document.write(html)
   342→   342→          printWindow.document.close()
   343→   343→          setTimeout(() => {
   344→   344→            printWindow.print()
   345→   345→          }, 500)
   346→   346→        }
   347→   347→        toast({ title: 'PDF Ready!', description: 'Use the print dialog to save as PDF.' })
   348→   348→      } else {
   349→   349→        toast({ title: 'Failed to generate PDF', variant: 'destructive' })
   350→   350→      }
   351→   351→    } catch {
   352→   352→      toast({ title: 'Error exporting PDF', variant: 'destructive' })
   353→   353→    }
   354→   354→  }
   355→   355→
   356→   356→  const formatPrice = (price: number) => {
   357→   357→    return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 0 }).format(price)
   358→   358→  }
   359→   359→
   360→   360→  if (loading) {
   361→   361→    return (
   362→   362→      <div className="flex items-center justify-center h-64">
   363→   363→        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-amber-500" />
   364→   364→      </div>
   365→   365→    )
   366→   366→  }
   367→   367→
   368→   368→  return (
   369→   369→    <div className="space-y-6">
   370→   370→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   371→   371→        <div>
   372→   372→          <h1 className="text-2xl font-bold text-slate-900 flex items-center gap-2">
   373→   373→            <FileText className="h-6 w-6 text-amber-500" />
   374→   374→            AI Website Proposal
   375→   375→          </h1>
   376→   376→          <p className="text-muted-foreground">Generate professional 3-tier proposals for businesses</p>
   377→   377→        </div>
   378→   378→      </div>
   379→   379→
   380→   380→      {/* Stats */}
   381→   381→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   382→   382→        <Card className="border-0 shadow-sm">
   383→   383→          <CardContent className="p-4">
   384→   384→            <div className="flex items-center gap-3">
   385→   385→              <div className="h-10 w-10 rounded-xl bg-slate-50 flex items-center justify-center">
   386→   386→                <Building2 className="h-5 w-5 text-slate-600" />
   387→   387→              </div>
   388→   388→              <div>
   389→   389→                <p className="text-2xl font-bold text-slate-900">{businesses.length}</p>

   391→   391→              </div>
   392→   392→            </div>
   393→   393→          </CardContent>
   394→   394→        </Card>
   395→   395→        <Card className="border-0 shadow-sm">
   396→   396→          <CardContent className="p-4">
   397→   397→            <div className="flex items-center gap-3">

   417→   417→              </div>

   419→   419→          </CardContent>
   420→   420→        </Card>
   421→   421→        <Card className="border-0 shadow-sm">
   422→   422→          <CardContent className="p-4">
   423→   423→            <div className="flex items-center gap-3">
   424→   424→              <div className="h-10 w-10 rounded-xl bg-red-50 flex items-center justify-center">

   434→   434→      </div>
   435→   435→
   436→   436→      {/* Filters */}
   437→   437→      <Card className="border-0 shadow-sm">
   438→   438→        <CardContent className="p-4">
   439→   439→          <div className="flex flex-col sm:flex-row gap-4">
