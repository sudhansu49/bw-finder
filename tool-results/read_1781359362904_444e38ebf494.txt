   158→function formatCurrency(value: number): string {
   159→  if (value >= 1000000) return `$${(value / 1000000).toFixed(1)}M`
   160→  if (value >= 1000) return `$${(value / 1000).toFixed(1)}k`
   161→  return `$${value.toLocaleString()}`
   162→}
   163→
   164→function formatDate(dateStr: string): string {
   165→  return new Date(dateStr).toLocaleDateString('en-US', {
   166→    month: 'short',
   167→    day: 'numeric',
   168→    year: 'numeric',
   169→  })
   170→}
   171→
   172→function getInitials(name: string): string {
   173→  return name
   174→    .split(' ')
   175→    .map((n) => n[0])
   176→    .join('')
   177→    .toUpperCase()
   178→    .slice(0, 2)
   179→}

   181→function statusBadgeClass(status: string): string {
   182→  switch (status) {
   183→    case 'active':
   184→      return 'bg-emerald-50 text-emerald-700 border-emerald-200'
   185→    case 'suspended':
   186→      return 'bg-amber-50 text-amber-700 border-amber-200'
   187→    case 'banned':
   188→      return 'bg-red-50 text-red-700 border-red-200'
   189→    case 'canceled':
   190→      return 'bg-red-50 text-red-600 border-red-200'
   191→    case 'expired':
   192→      return 'bg-slate-50 text-slate-600 border-slate-200'
   193→    case 'past_due':
   194→      return 'bg-rose-50 text-rose-700 border-rose-200'
   195→    default:
   196→      return 'bg-slate-50 text-slate-600 border-slate-200'
   197→  }
   198→}
