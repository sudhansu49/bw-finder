   152→function getScoreColor(score: number | null | undefined): { bg: string; text: string } {
   153→  if (score === null || score === undefined) return { bg: 'bg-slate-100', text: 'text-slate-400' }
   154→  if (score >= 70) return { bg: 'bg-emerald-50', text: 'text-emerald-600' }
   155→  if (score >= 40) return { bg: 'bg-amber-50', text: 'text-amber-600' }
   156→  return { bg: 'bg-red-50', text: 'text-red-500' }
   157→}
